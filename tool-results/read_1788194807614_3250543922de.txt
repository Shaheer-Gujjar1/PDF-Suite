   138→/** Compute size/position of one element on the FINAL canvas — mirrors
   139→ *  drawMemeText in worker-source.ts exactly. */
   140→function layoutMemeElement(
   141→  ctx: CanvasRenderingContext2D,
   142→  W: number,
   143→  H: number,
   144→  el: MemeTextElement
   145→): ElementLayout | null {
   146→  const raw = el.text || ''
   147→  if (!raw.replace(/\s/g, '')) return null
   148→  const txt = el.caps ? raw.toUpperCase() : raw
   149→  const lines = txt.split('\n')
   150→  let px: number
   151→  if (!el.sizePx || el.sizePx <= 0) {
   152→    px = Math.max(14, Math.round(W / 8))
   153→    ctx.font = buildMemeFont(el, px)
   154→    let maxw = 0
   155→    for (const line of lines) maxw = Math.max(maxw, ctx.measureText(line).width)
   156→    if (maxw > W * MEME_FIT_W && maxw > 0) px = Math.max(12, Math.floor(px * ((W * MEME_FIT_W) / maxw)))
   157→  } else {
   158→    px = el.sizePx
   159→  }
   160→  ctx.font = buildMemeFont(el, px)
   161→  ctx.textAlign = el.align || 'center'
   162→  ctx.textBaseline = 'alphabetic'
   163→  const lh = px * MEME_LINE_H
   164→  const blockH = lines.length * lh
   165→  const ax = el.x * W
   166→  const y0 = el.y * H - blockH / 2 + lh / 2 + px * 0.35
   167→  const boxes: ElementLayout['boxes'] = []
   168→  for (let i = 0; i < lines.length; i++) {
   169→    const line = lines[i]
   170→    if (!line) continue
   171→    const w = ctx.measureText(line).width
   172→    const lx = el.align === 'left' ? ax : el.align === 'right' ? ax - w : ax - w / 2
   173→    boxes.push({ x: lx - px * 0.1, y: y0 + i * lh - px * 0.9, w: w + px * 0.2, h: lh })
   174→  }
   175→  const bbox = boxes.length
   176→    ? {
   177→        x: Math.min(...boxes.map((b) => b.x)),
   178→        y: Math.min(...boxes.map((b) => b.y)),
   179→        w: Math.max(...boxes.map((b) => b.x + b.w)) - Math.min(...boxes.map((b) => b.x)),
   180→        h: Math.max(...boxes.map((b) => b.y + b.h)) - Math.min(...boxes.map((b) => b.y)),
   181→      }
   182→    : { x: ax - px, y: y0 - px, w: px * 2, h: blockH }
   183→  return { px, boxes, bbox }
   184→}
   185→
   186→function paintMemeElement(
   187→  ctx: CanvasRenderingContext2D,
   188→  W: number,
   189→  H: number,
   190→  el: MemeTextElement
   191→): ElementLayout | null {
   192→  const lay = layoutMemeElement(ctx, W, H, el)
   193→  if (!lay) return null
   194→  const raw = el.text || ''
   195→  const txt = el.caps ? raw.toUpperCase() : raw
   196→  const lines = txt.split('\n')
   197→  const lh = lay.px * MEME_LINE_H
   198→  const ax = el.x * W
   199→  const y0 = el.y * H - lines.length * lh / 2 + lh / 2 + lay.px * 0.35
   200→  const sw = (el.strokeWidth || 0) * lay.px
   201→  ctx.font = buildMemeFont(el, lay.px)
   202→  ctx.textAlign = el.align || 'center'
   203→  ctx.textBaseline = 'alphabetic'
   204→  for (let i = 0; i < lines.length; i++) {
   205→    const ly = y0 + i * lh
   206→    if (!lines[i]) continue
   207→    if (sw > 0) {
   208→      ctx.lineJoin = 'round'
   209→      ctx.miterLimit = 2
   210→      ctx.lineWidth = sw
   211→      ctx.strokeStyle = el.strokeColor || '#000000'
   212→      ctx.strokeText(lines[i], ax, ly)
   213→    }
   214→    ctx.fillStyle = el.color || '#ffffff'
   215→    ctx.fillText(lines[i], ax, ly)
   216→    if (el.underline) {
   217→      const wU = ctx.measureText(lines[i]).width
   218→      const ux = el.align === 'left' ? ax : el.align === 'right' ? ax - wU : ax - wU / 2
   219→      ctx.fillRect(ux, ly + lay.px * 0.12, wU, Math.max(2, lay.px * 0.06))
   220→    }
   221→  }
   222→  return lay
   223→}

   228→function MemePreview({
   229→  file,
   230→  state,
   231→  activeElementId,
   232→  onSelect,
   233→  onMove,
   234→  onMetrics,
   235→}: {
   236→  file: File
   237→  state: MemeState
   238→  activeElementId: string | null
   239→  onSelect: (id: string | null) => void
   240→  onMove: (id: string, x: number, y: number) => void
   241→  onMetrics: (m: Record<string, number>) => void
   242→}) {
   243→  const canvasRef = React.useRef<HTMLCanvasElement>(null)
   244→  const [baseImg, setBaseImg] = React.useState<HTMLImageElement | null>(null)
   245→  const layoutsRef = React.useRef<Record<string, ElementLayout>>({})
   246→  const dragRef = React.useRef<{ id: string; dx: number; dy: number } | null>(null)
   247→
   248→  /* Load the base image. */
   249→  React.useEffect(() => {
   250→    let cancelled = false
   251→    setBaseImg(null)
   252→    const url = URL.createObjectURL(file)
   253→    const img = new Image()
   254→    img.onload = () => {
   255→      if (!cancelled) setBaseImg(img)
   256→    }
   257→    img.onerror = () => {
   258→      if (!cancelled) setBaseImg(null)
   259→    }
   260→    img.src = url
   261→    return () => {
   262→      cancelled = true
   263→      URL.revokeObjectURL(url)
   264→    }
   265→  }, [file])
   266→
   267→  /* Redraw on any change. */
   268→  React.useEffect(() => {
   269→    const canvas = canvasRef.current
   270→    if (!canvas || !baseImg) return
   271→    const ctx = canvas.getContext('2d')
   272→    if (!ctx) return
   273→    const W = baseImg.naturalWidth || 1
   274→    const imgH = baseImg.naturalHeight || 1
   275→    const bar = state.mode === 'outside' ? Math.round(imgH * MEME_BAR_FRAC) : 0
   276→    const H = imgH + bar * 2
   277→    canvas.width = W
   278→    canvas.height = H
   279→    ctx.clearRect(0, 0, W, H)
   280→    if (bar > 0 || file.type === 'image/jpeg') {
   281→      ctx.fillStyle = '#ffffff'
   282→      ctx.fillRect(0, 0, W, H)
   283→    }
   284→    ctx.drawImage(baseImg, 0, bar)
   285→
   286→    const layouts: Record<string, ElementLayout> = {}
   287→    const metrics: Record<string, number> = {}
   288→    for (const el of state.elements) {
   289→      const lay = paintMemeElement(ctx, W, H, el)
   290→      if (lay) {
   291→        layouts[el.id] = lay
   292→        metrics[el.id] = lay.px
   293→      }
   294→    }
   295→    layoutsRef.current = layouts
   296→
   297→    /* Selection outline for the active element (preview only). */
   298→    const active = activeElementId ? layouts[activeElementId] : null
   299→    if (active) {
   300→      ctx.save()
   301→      ctx.setLineDash([6, 4])
   302→      ctx.lineWidth = Math.max(2, Math.round(W / 400))
   303→      ctx.strokeStyle = 'rgba(139, 92, 246, 0.95)'
   304→      ctx.strokeRect(
   305→        active.bbox.x - 6,
   306→        active.bbox.y - 6,
   307→        active.bbox.w + 12,
   308→        active.bbox.h + 12
   309→      )
   310→      ctx.restore()
   311→    }
   312→    onMetrics(metrics)
   313→  }, [baseImg, state, activeElementId, file, onMetrics])
   314→
   315→  const toCanvas = (e: React.PointerEvent<HTMLCanvasElement>) => {
   316→    const canvas = canvasRef.current!
   317→    const rect = canvas.getBoundingClientRect()
   318→    return {
   319→      cx: ((e.clientX - rect.left) * canvas.width) / rect.width,
   320→      cy: ((e.clientY - rect.top) * canvas.height) / rect.height,
   321→    }
   322→  }
   323→
   324→  const hitTest = (cx: number, cy: number) => {
   325→    const pad = 10
   326→    const entries = state.elements.filter((el) => layoutsRef.current[el.id])
   327→    for (let i = entries.length - 1; i >= 0; i--) {
   328→      const el = entries[i]
   329→      const lay = layoutsRef.current[el.id]
   330→      const inBox =
   331→        cx >= lay.bbox.x - pad &&
   332→        cx <= lay.bbox.x + lay.bbox.w + pad &&
   333→        cy >= lay.bbox.y - pad &&
   334→        cy <= lay.bbox.y + lay.bbox.h + pad
   335→      if (inBox) return el
   336→    }
   337→    return null
   338→  }
   339→
   340→  return (
   341→    <div className="overflow-hidden rounded-2xl border border-border/60" style={CHECKER_BG}>
   342→      {baseImg ? (
   343→        <canvas
   344→          ref={canvasRef}
   345→          className="mx-auto block h-auto max-h-[560px] w-auto max-w-full cursor-move touch-none select-none"
   346→          onPointerDown={(e) => {
   347→            const { cx, cy } = toCanvas(e)
   348→            const hit = hitTest(cx, cy)
   349→            if (hit) {
   350→              onSelect(hit.id)
   351→              const canvas = canvasRef.current!
   352→              dragRef.current = {
   353→                id: hit.id,
   354→                dx: cx - hit.x * canvas.width,
   355→                dy: cy - hit.y * canvas.height,
   356→              }
   357→              canvas.setPointerCapture(e.pointerId)
   358→            } else {
   359→              onSelect(null)
   360→            }
   361→          }}
   362→          onPointerMove={(e) => {
   363→            const drag = dragRef.current
   364→            if (!drag) return
   365→            const { cx, cy } = toCanvas(e)
   366→            const canvas = canvasRef.current!
   367→            const nx = Math.min(0.98, Math.max(0.02, (cx - drag.dx) / canvas.width))
   368→            const ny = Math.min(0.99, Math.max(0.01, (cy - drag.dy) / canvas.height))
   369→            onMove(drag.id, nx, ny)
   370→          }}
   371→          onPointerUp={() => {
   372→            dragRef.current = null
   373→          }}
   374→          onPointerCancel={() => {
   375→            dragRef.current = null
   376→          }}
   377→        />
   378→      ) : (
   379→        <div className="grid h-64 place-items-center">
   380→          <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   381→        </div>
   382→      )}
   383→    </div>
   384→  )
   385→}

   589→    const missing = files.filter((f) => !urlsRef.current[f.id])

   592→      urlsRef.current[f.id] = url
   593→      const img = new Image()
   594→      img.onload = () => {
   595→        if (cancelled) return
   596→        setMeta((prev) => ({
   597→          ...prev,
   598→          [f.id]: {
   599→            url,
   600→            width: img.naturalWidth || 1,
   601→            height: img.naturalHeight || 1,
   602→          },
   603→        }))
   604→      }
   605→      img.onerror = () => {
   606→        if (cancelled) return
   607→        setMeta((prev) => ({ ...prev, [f.id]: { url, width: 1, height: 1 } }))
   608→      }
   609→      img.src = url
   610→    }
   611→
   612→    return () => {
   613→      cancelled = true
   614→    }
   615→  }, [files])
   616→
   617→  // Revoke everything on unmount.
   618→  React.useEffect(() => {

   620→    return () => {
   621→      for (const id of Object.keys(urls)) URL.revokeObjectURL(urls[id])
   622→    }
   623→  }, [])
   624→
   625→  /* ---------------- Initialize state for new files ---------------------- */

   631→        if (!next[f.id]) {

   634→        }
   635→      }
   636→      for (const id of Object.keys(next)) {
   637→        if (!files.some((f) => f.id === id)) {
   638→          delete next[id]
   639→          changed = true
   640→        }
   641→      }
   642→      return changed ? next : prev
   643→    })
   644→  }, [files])
   645→
   646→  /* Keep the active image valid. */
   647→  React.useEffect(() => {

   649→      setActiveImageId(null)
   650→      return
   651→    }
   652→    setActiveImageId((prev) =>
   653→      prev && files.some((f) => f.id === prev) ? prev : files[0].id
   654→    )
   655→  }, [files])
   656→
   657→  const activeFile = files.find((f) => f.id === activeImageId) ?? files[0] ?? null

   659→    ? memes[activeFile.id] ?? makeMemeState()
   660→    : null
   661→  const activeElement =
   662→    activeState?.elements.find((e) => e.id === activeElementId) ?? null
   663→
   664→  /* Keep the active element valid (per image). */
   665→  React.useEffect(() => {
   666→    if (!activeFile) return
   667→    const st = memes[activeFile.id]
   668→    if (!st) return
   669→    if (activeElementId && st.elements.some((e) => e.id === activeElementId)) return

   672→
   673→  /* ---------------- Emit result whenever memes change -------------------- */
   674→  React.useEffect(() => {
   675→    if (files.length === 0) {
   676→      onChange(null)
   677→      return
   678→    }
   679→    const result: Record<string, MemeState> = {}
   680→    for (const f of files) {
   681→      result[f.file.name] = memes[f.id] ?? makeMemeState()
   682→    }
   683→    onChange({ memes: result })
   684→  }, [files, memes, onChange])
   685→
   686→  /* ---------------- Mutators --------------------------------------------- */
   687→  const patchActive = (patch: Partial<MemeState>) => {
   688→    if (!activeFile) return
   689→    setMemes((prev) => {
   690→      const cur = prev[activeFile.id] ?? makeMemeState()

   692→    })
   693→  }
   694→

   698→      const cur = prev[activeFile.id] ?? makeMemeState()

   700→        ...prev,
   701→        [activeFile.id]: {
   702→          ...cur,
   703→          elements: cur.elements.map((e) => (e.id === id ? { ...e, ...patch } : e)),
   704→        },
   705→      }
   706→    })
   707→  }

   714→    const el = makeElement({ text: 'New text', y: 0.5 })

   716→    setActiveElementId(el.id)
   717→  }
   718→
   719→  const duplicateElement = () => {
   720→    if (!activeElement || !activeState) return
   721→    const el = makeElement({
   722→      ...activeElement,
   723→      id: uid('el'),
   724→      y: Math.min(0.95, activeElement.y + 0.08),
   725→    })
   726→    patchActive({ elements: [...activeState.elements, el] })
   727→    setActiveElementId(el.id)
   728→  }
   729→
   730→  const removeElement = (id: string) => {
   731→    if (!activeFile || !activeState) return
   732→    patchActive({ elements: activeState.elements.filter((e) => e.id !== id) })
   733→  }
   734→
   735→  /** Inside ↔ Outside: remap every element's y so text stays glued to the
   736→   *  image content (bars are added/removed around it). */
   737→  const toggleMode = (mode: 'inside' | 'outside') => {
   738→    if (!activeFile || !activeState || activeState.mode === mode) return
   739→    const imgH = (meta[activeFile.id]?.height ?? 1) || 1
   740→    const bar = Math.round(imgH * MEME_BAR_FRAC)
   741→    const total = imgH + bar * 2
   742→    const elements = activeState.elements.map((el) => {
   743→      let y = el.y
   744→      if (mode === 'outside') y = (bar + el.y * imgH) / total
   745→      else y = (el.y * total - bar) / imgH
   746→      return { ...el, y: Math.min(0.99, Math.max(0.01, y)) }
   747→    })
   748→    patchActive({ mode, elements })
   749→  }
   750→
   751→  const elementLabel = (el: MemeTextElement, i: number) =>
   752→    `Text ${i + 1} · ${el.text.trim().replace(/\s+/g, ' ').slice(0, 16) || 'empty'}`
   753→
   754→  const handleMetrics = React.useCallback((m: Record<string, number>) => {
   755→    setMetrics((prev) => {
   756→      const keys = new Set([...Object.keys(prev), ...Object.keys(m)])
   757→      for (const k of keys) if (prev[k] !== m[k]) return m
   758→      return prev
   759→    })
   760→  }, [])
   761→
   762→  const effectivePx = activeElement ? metrics[activeElement.id] ?? 0 : 0
   763→
   764→  return (
   765→    <div className="space-y-4">
   766→      <div className="grid gap-4 lg:grid-cols-[minmax(0,1fr)_360px]">
   767→        {/* ------------------------- Canvas column ------------------------- */}
   768→        <div className="space-y-3">
   769→          {/* Floating text toolbar (iloveimg-style pill, ABOVE the canvas so
   770→              it never blocks dragging the top caption) */}
   771→          {activeElement && (
   772→            <div className="flex justify-center">
   773→              <div className="flex items-center gap-1 rounded-xl border border-border/70 bg-card/95 p-1 shadow-lg backdrop-blur">
   774→                <Select
   775→                  value={activeElement.font}
   776→                  onValueChange={(v) => patchElement(activeElement.id, { font: v })}
   777→                >
   778→                  <SelectTrigger
   779→                    className="h-8 w-[120px] text-xs"
   780→                    aria-label="Font family"
   781→                  >
   782→                    <SelectValue />
   783→                  </SelectTrigger>
   784→                  <SelectContent>
   785→                    {MEME_FONTS.map((f) => (
   786→                      <SelectItem key={f} value={f}>
   787→                        {f}
   788→                      </SelectItem>
   789→                    ))}
   790→                  </SelectContent>
   791→                </Select>
   792→                {(
   793→                  [
   794→                    ['bold', <Bold key="b" className="h-3.5 w-3.5" />, 'Bold'],

   796→                    ['underline', <Underline key="u" className="h-3.5 w-3.5" />, 'Underline'],
   797→                    ['caps', <CaseUpper key="c" className="h-3.5 w-3.5" />, 'ALL-CAPS'],
   798→                  ] as const
   799→                ).map(([key, icon, label]) => (
   800→                  <button
   801→                    key={key}
   802→                    type="button"
   803→                    onClick={() => patchElement(activeElement.id, { [key]: !activeElement[key] })}
   804→                    aria-label={`${label} toggle`}
   805→                    aria-pressed={activeElement[key]}
   806→                    className={cn(
   807→                      'grid h-8 w-8 place-items-center rounded-lg text-muted-foreground transition-colors hover:bg-secondary hover:text-foreground',
   808→                      activeElement[key] && 'bg-primary/10 text-primary hover:bg-primary/10 hover:text-primary'
   809→                    )}
   810→                  >
   811→                    {icon}
   812→                  </button>
   813→                ))}
   814→                <div className="mx-0.5 h-5 w-px bg-border" />
   815→                {(['left', 'center', 'right'] as const).map((a) => {
   816→                  const Icon = a === 'left' ? AlignLeft : a === 'center' ? AlignCenter : AlignRight
   817→                  return (
   818→                    <button
   819→                      key={a}
   820→                      type="button"
   821→                      onClick={() => patchElement(activeElement.id, { align: a })}
   822→                      aria-label={`Align ${a}`}
   823→                      aria-pressed={activeElement.align === a}
   824→                      className={cn(
   825→                        'grid h-8 w-8 place-items-center rounded-lg text-muted-foreground transition-colors hover:bg-secondary hover:text-foreground',
   826→                        activeElement.align === a && 'bg-primary/10 text-primary hover:bg-primary/10 hover:text-primary'
   827→                      )}
   828→                    >
   829→                      <Icon className="h-3.5 w-3.5" />
   830→                    </button>
   831→                  )
   832→                })}
   833→                <div className="mx-0.5 h-5 w-px bg-border" />
   834→                <button
   835→                  type="button"
   836→                  onClick={duplicateElement}
   837→                  aria-label="Duplicate this text"
   838→                  className="grid h-8 w-8 place-items-center rounded-lg text-muted-foreground transition-colors hover:bg-secondary hover:text-foreground"

   840→                  <Copy className="h-3.5 w-3.5" />

   847→                >
   848→                  <Trash2 className="h-3.5 w-3.5" />
   849→                </button>
   850→              </div>
   851→            </div>

   853→
   854→          <div className="relative">

   859→                state={activeState}
   860→                activeElementId={activeElementId}
   861→                onSelect={setActiveElementId}
   862→                onMove={moveElement}

   866→              <div
   867→                className="grid h-64 place-items-center rounded-2xl border border-border/60 text-sm text-muted-foreground"

   870→                No image selected
   871→              </div>
   872→            )}
   873→
   874→            {/* Floating add-text button */}
   875→            <div className="absolute right-4 top-1/2 z-20 -translate-y-1/2">

   877→                type="button"
   878→                onClick={addElement}
   879→                aria-label="Add a text box"
   880→                className="relative grid h-12 w-12 place-items-center rounded-full bg-primary text-primary-foreground shadow-lg transition-transform hover:scale-105"
   881→              >
   882→                <Plus className="h-5 w-5" />
   883→                {(activeState?.elements.length ?? 0) > 0 && (
   884→                  <span className="absolute -left-1.5 -top-1.5 grid h-5 min-w-5 place-items-center rounded-full bg-foreground px-1 text-[10px] font-bold text-background">
   885→                    {activeState?.elements.length}
   886→                  </span>
   887→                )}
   888→              </button>
   889→            </div>
   890→          </div>
   891→
   892→          {/* Queued images film strip */}
   893→          <div className="flex gap-2 overflow-x-auto pb-1">
   894→            {files.map((f) => {
   895→              const active = f.id === activeFile?.id
   896→              const url = meta[f.id]?.url
   897→              return (
   898→                <div key={f.id} className="group relative shrink-0">
   899→                  <button
   900→                    type="button"
   901→                    onClick={() => setActiveImageId(f.id)}
   902→                    aria-label={`Edit meme for ${f.file.name}`}
   903→                    aria-pressed={active}
   904→                    className={cn(
   905→                      'block h-14 w-20 overflow-hidden rounded-lg border-2 bg-muted transition-all',
   906→                      active
   907→                        ? 'border-primary'
   908→                        : 'border-transparent opacity-80 hover:opacity-100'
   909→                    )}
   910→                  >
   911→                    {url ? (
   912→                      <img
   913→                        src={url}
   914→                        alt=""
   915→                        className="h-full w-full object-cover"
   916→                        draggable={false}
   917→                      />
   918→                    ) : (
   919→                      <Loader2 className="mx-auto mt-5 h-4 w-4 animate-spin text-muted-foreground" />
   920→                    )}
   921→                  </button>
   922→                  <button
   923→                    type="button"
   924→                    onClick={() => onRemove(f.id)}
   925→                    aria-label={`Remove ${f.file.name}`}
   926→                    className="absolute -right-1.5 -top-1.5 hidden h-5 w-5 place-items-center rounded-full bg-destructive text-white shadow group-hover:grid"
   927→                  >
   928→                    <X className="h-3 w-3" />
   929→                  </button>
   930→                </div>
   931→              )
   932→            })}
   933→            <button
   934→              type="button"
   935→              onClick={onAddMore}
   936→              aria-label="Add more images"
   937→              className="grid h-14 w-20 shrink-0 place-items-center rounded-lg border-2 border-dashed border-border text-muted-foreground transition-colors hover:border-primary/50 hover:text-primary"
   938→            >
   939→              <ImagePlus className="h-4 w-4" />
   940→            </button>
   941→          </div>
   942→        </div>
   943→
   944→        {/* ------------------------- Sidebar column ------------------------ */}
   945→        <div className="space-y-3">
   946→          <div className="rounded-2xl border border-border/70 bg-secondary/40 p-4">
   947→            <p className="text-sm font-semibold">Meme editor</p>

   949→            {/* Inside / Outside toggle */}
   950→            <div className="mt-3 grid grid-cols-2 gap-2">

   953→                  ['inside', 'Text inside'],
   954→                  ['outside', 'Text outside'],
   955→                ] as const

   957→                <button
   958→                  key={m}
   959→                  type="button"
   960→                  onClick={() => toggleMode(m)}
   961→                  aria-pressed={activeState?.mode === m}
   962→                  className={cn(
   963→                    'rounded-xl border px-3 py-2.5 text-xs font-medium transition-colors',
   964→                    activeState?.mode === m
   965→                      ? 'border-primary bg-primary/5 text-foreground'
   966→                      : 'border-border bg-card text-muted-foreground hover:border-primary/40'
   967→                  )}
   968→                >
   969→                  {label}
   970→                </button>
   971→              ))}
   972→            </div>
   973→            <p className="mt-1.5 text-[11px] leading-snug text-muted-foreground">
   974→              &ldquo;Outside&rdquo; adds classic white bars above and below the
   975→              image; your text stays glued to the picture.
   976→            </p>
   977→
   978→            <div className="mt-3 grid gap-2">
   979→              <button
   980→                type="button"
   981→                onClick={onAddMore}
   982→                className="flex items-center gap-3 rounded-xl border border-border bg-card px-3 py-2.5 text-sm font-medium transition-colors hover:border-primary/50 hover:bg-primary/5"

   994→                <span className="grid h-8 w-8 place-items-center rounded-lg bg-primary text-primary-foreground">
   995→                  <Type className="h-4 w-4" />
   996→                </span>
   997→                Add text
   998→              </button>
   999→            </div>
  1000→
  1001→            {activeState && activeState.elements.length > 0 && (
  1002→              <div className="mt-3 space-y-1.5">
  1003→                {activeState.elements.map((el, i) => {

  1009→                      tabIndex={0}
  1010→                      onClick={() => setActiveElementId(el.id)}

  1034→                      >
  1035→                        <Trash2 className="h-3 w-3" />
  1036→                      </button>

  1038→                  )

  1041→            )}
  1042→          </div>
  1043→
  1044→          {activeElement ? (
  1045→            <>
  1046→              <ElementCard
  1047→                el={activeElement}
  1048→                effectivePx={effectivePx}
  1049→                onChange={(patch) => patchElement(activeElement.id, patch)}
  1050→                onRemove={() => removeElement(activeElement.id)}

  1052→              <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
  1053→                <span className="inline-block h-1.5 w-1.5 animate-pulse rounded-full bg-emerald-500" />
  1054→                Editing live — drag text on the canvas or type here; every
  1055→                change updates the preview instantly.
  1056→              </p>

  1058→          ) : (
  1059→            <div className="rounded-2xl border border-dashed border-border bg-card/60 p-5 text-center">

  1072→        <Laugh className="mt-0.5 h-4 w-4 shrink-0 text-violet-500" />
  1073→        <p className="text-muted-foreground">
  1074→          Pick an image from the film strip, then click any text on the canvas

  1076→          caps and duplicate; the sidebar has fonts, sizes and colors. What

  1079→        </p>
  1080→      </div>
  1081→    </div>
