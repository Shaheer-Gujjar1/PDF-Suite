    65→    65→interface InputConfig {
    66→    66→  accept: string
    67→    67→  multiple: boolean
    68→    68→  hint: string
    69→    69→  mode: 'files' | 'text'

    72→    72→function getInput(tool: Tool): InputConfig {
    73→    73→  switch (tool.id) {
    74→    74→    case 'crop-images':
    75→    75→      return { accept: 'image/jpeg,image/jpg,image/png,image/webp', multiple: true, hint: 'JPG, PNG, WEBP images', mode: 'files' }
    76→    76→    case 'convert-images':
    77→    77→      return { accept: 'image/*', multiple: true, hint: 'JPG, PNG, WEBP, GIF, BMP — any image format', mode: 'files' }
    78→    78→    case 'favicon-generator':
    79→    79→      return { accept: 'image/*', multiple: true, hint: 'JPG, PNG, WEBP, GIF, BMP — any image format', mode: 'files' }
    80→    80→    case 'watermark-images':
    81→    81→      return { accept: 'image/*', multiple: true, hint: 'JPG, PNG, WEBP, GIF, BMP — any image format', mode: 'files' }
    82→    82→    case 'images-to-pdf':
    83→    83→      return { accept: 'image/*', multiple: true, hint: 'JPG, PNG, WEBP, GIF', mode: 'files' }
    84→    84→    case 'word-to-pdf':
    85→    85→      return { accept: '.docx,application/vnd.openxmlformats-officedocument.wordprocessingml.document', multiple: true, hint: 'Word .docx files', mode: 'files' }
    86→    86→    case 'excel-to-pdf':
    87→    87→      return { accept: '.xlsx,application/vnd.openxmlformats-officedocument.spreadsheetml.sheet', multiple: true, hint: 'Excel .xlsx files', mode: 'files' }
    88→    88→    case 'html-to-pdf':
    89→    89→      return { accept: '', multiple: false, hint: 'Paste your HTML', mode: 'text' }
    90→    90→    default:
    91→    91→      return { accept: 'application/pdf', multiple: tool.batch, hint: 'PDF files', mode: 'files' }
    92→    92→  }
    93→    93→}

   360→   360→      singleLabel = 'HTML → PDF output'

   364→   364→      mode = 'single'
   365→   365→      singleLabel = 'HTML → PDF output'
   366→   366→    } else if (isWordToPdf) {

   370→   370→      const orderedFiles = wordFiles.map((wf) => files.find((f) => f.id === wf.id)!).filter(Boolean)

   372→   372→      for (const qf of orderedFiles) {
   373→   373→        try {
   374→   374→          const pages = await renderDocxToPages(qf.file)
   375→   375→          if (pages.length === 0) throw new Error('No pages rendered')
   376→   376→          for (const page of pages) {
   377→   377→            // Convert data URL to ArrayBuffer
   378→   378→            const base64 = page.dataUrl.split(',')[1]
   379→   379→            const binary = atob(base64)
   380→   380→            const arr = new Uint8Array(binary.length)
   381→   381→            for (let j = 0; j < binary.length; j++) arr[j] = binary.charCodeAt(j)
   382→   382→            // Verify the JPEG starts with the correct magic bytes (0xFF 0xD8)
   383→   383→            if (arr[0] !== 0xff || arr[1] !== 0xd8) {
   384→   384→              throw new Error('Invalid JPEG data')
   385→   385→            }
   386→   386→            inputs.push({
   387→   387→              fileName: 'page-' + (inputs.length + 1) + '.jpg',
   388→   388→              data: arr.buffer,
   389→   389→              size: arr.buffer.byteLength,
   390→   390→            })
   391→   391→          }
   392→   392→        } catch (e) {

   942→   942→      )}
   943→   943→    </div>
   944→   944→  )

   947→   947→function InteractiveEditor({
   948→   948→  toolId,

   951→   951→  onRemoveFile,
