   132→   132→interface ElementLayout {
   133→   133→  px: number
   134→   134→  boxes: { x: number; y: number; w: number; h: number }[]
   135→   135→  bbox: { x: number; y: number; w: number; h: number }

   138→   138→/** Compute size/position of one element on the FINAL canvas — mirrors
   139→   139→ *  drawMemeText in worker-source.ts exactly. */
   140→   140→function layoutMemeElement(
   141→   141→  ctx: CanvasRenderingContext2D,
   142→   142→  W: number,
   143→   143→  H: number,
   144→   144→  el: MemeTextElement
   145→   145→): ElementLayout | null {
   146→   146→  const raw = el.text || ''
   147→   147→  if (!raw.replace(/\s/g, '')) return null
   148→   148→  const txt = el.caps ? raw.toUpperCase() : raw
   149→   149→  const lines = txt.split('\n')
   150→   150→  let px: number
   151→   151→  if (!el.sizePx || el.sizePx <= 0) {
   152→   152→    px = Math.max(14, Math.round(W / 8))
   153→   153→    ctx.font = buildMemeFont(el, px)
   154→   154→    let maxw = 0
   155→   155→    for (const line of lines) maxw = Math.max(maxw, ctx.measureText(line).width)
   156→   156→    if (maxw > W * MEME_FIT_W && maxw > 0) px = Math.max(12, Math.floor(px * ((W * MEME_FIT_W) / maxw)))
   157→   157→  } else {
   158→   158→    px = el.sizePx
   159→   159→  }
   160→   160→  ctx.font = buildMemeFont(el, px)
   161→   161→  ctx.textAlign = el.align || 'center'
   162→   162→  ctx.textBaseline = 'alphabetic'
   163→   163→  const lh = px * MEME_LINE_H
   164→   164→  const blockH = lines.length * lh
   165→   165→  const ax = el.x * W
   166→   166→  const y0 = el.y * H - blockH / 2 + lh / 2 + px * 0.35
   167→   167→  const boxes: ElementLayout['boxes'] = []
   168→   168→  for (let i = 0; i < lines.length; i++) {
   169→   169→    const line = lines[i]
   170→   170→    if (!line) continue
   171→   171→    const w = ctx.measureText(line).width
   172→   172→    const lx = el.align === 'left' ? ax : el.align === 'right' ? ax - w : ax - w / 2
   173→   173→    boxes.push({ x: lx - px * 0.1, y: y0 + i * lh - px * 0.9, w: w + px * 0.2, h: lh })
   174→   174→  }
   175→   175→  const bbox = boxes.length
   176→   176→    ? {
   177→   177→        x: Math.min(...boxes.map((b) => b.x)),
   178→   178→        y: Math.min(...boxes.map((b) => b.y)),
   179→   179→        w: Math.max(...boxes.map((b) => b.x + b.w)) - Math.min(...boxes.map((b) => b.x)),
   180→   180→        h: Math.max(...boxes.map((b) => b.y + b.h)) - Math.min(...boxes.map((b) => b.y)),
   181→   181→      }
   182→   182→    : { x: ax - px, y: y0 - px, w: px * 2, h: blockH }
   183→   183→  return { px, boxes, bbox }
   184→   184→}
   185→   185→
   186→   186→function paintMemeElement(
   187→   187→  ctx: CanvasRenderingContext2D,
   188→   188→  W: number,
   189→   189→  H: number,
   190→   190→  el: MemeTextElement
   191→   191→): ElementLayout | null {
   192→   192→  const lay = layoutMemeElement(ctx, W, H, el)
   193→   193→  if (!lay) return null
   194→   194→  const raw = el.text || ''
   195→   195→  const txt = el.caps ? raw.toUpperCase() : raw
   196→   196→  const lines = txt.split('\n')
   197→   197→  const lh = lay.px * MEME_LINE_H
   198→   198→  const ax = el.x * W
   199→   199→  const y0 = el.y * H - lines.length * lh / 2 + lh / 2 + lay.px * 0.35
   200→   200→  const sw = (el.strokeWidth || 0) * lay.px
   201→   201→  ctx.font = buildMemeFont(el, lay.px)
   202→   202→  ctx.textAlign = el.align || 'center'
   203→   203→  ctx.textBaseline = 'alphabetic'
   204→   204→  for (let i = 0; i < lines.length; i++) {
   205→   205→    const ly = y0 + i * lh
   206→   206→    if (!lines[i]) continue
   207→   207→    if (sw > 0) {
   208→   208→      ctx.lineJoin = 'round'
   209→   209→      ctx.miterLimit = 2
   210→   210→      ctx.lineWidth = sw
   211→   211→      ctx.strokeStyle = el.strokeColor || '#000000'
   212→   212→      ctx.strokeText(lines[i], ax, ly)
   213→   213→    }
   214→   214→    ctx.fillStyle = el.color || '#ffffff'
   215→   215→    ctx.fillText(lines[i], ax, ly)
   216→   216→    if (el.underline) {
   217→   217→      const wU = ctx.measureText(lines[i]).width
   218→   218→      const ux = el.align === 'left' ? ax : el.align === 'right' ? ax - wU : ax - wU / 2
   219→   219→      ctx.fillRect(ux, ly + lay.px * 0.12, wU, Math.max(2, lay.px * 0.06))
   220→   220→    }
   221→   221→  }
   222→   222→  return lay
   223→   223→}

   228→   228→function MemePreview({
   229→   229→  file,
   230→   230→  state,
   231→   231→  activeElementId,
   232→   232→  onSelect,
   233→   233→  onMove,
   234→   234→  onMetrics,
   235→   235→}: {
   236→   236→  file: File
   237→   237→  state: MemeState
   238→   238→  activeElementId: string | null
   239→   239→  onSelect: (id: string | null) => void
   240→   240→  onMove: (id: string, x: number, y: number) => void
   241→   241→  onMetrics: (m: Record<string, number>) => void
   242→   242→}) {
   243→   243→  const canvasRef = React.useRef<HTMLCanvasElement>(null)
   244→   244→  const [baseImg, setBaseImg] = React.useState<HTMLImageElement | null>(null)
   245→   245→  const layoutsRef = React.useRef<Record<string, ElementLayout>>({})
   246→   246→  const dragRef = React.useRef<{ id: string; dx: number; dy: number } | null>(null)
   247→   247→
   248→   248→  /* Load the base image. */
   249→   249→  React.useEffect(() => {
   250→   250→    let cancelled = false
   251→   251→    setBaseImg(null)
   252→   252→    const url = URL.createObjectURL(file)
   253→   253→    const img = new Image()
   254→   254→    img.onload = () => {
   255→   255→      if (!cancelled) setBaseImg(img)
   256→   256→    }
   257→   257→    img.onerror = () => {
   258→   258→      if (!cancelled) setBaseImg(null)
   259→   259→    }
   260→   260→    img.src = url
   261→   261→    return () => {
   262→   262→      cancelled = true
   263→   263→      URL.revokeObjectURL(url)
   264→   264→    }
   265→   265→  }, [file])
   266→   266→
   267→   267→  /* Redraw on any change. */
   268→   268→  React.useEffect(() => {
   269→   269→    const canvas = canvasRef.current
   270→   270→    if (!canvas || !baseImg) return
   271→   271→    const ctx = canvas.getContext('2d')
   272→   272→    if (!ctx) return
   273→   273→    const W = baseImg.naturalWidth || 1
   274→   274→    const imgH = baseImg.naturalHeight || 1
   275→   275→    const bar = state.mode === 'outside' ? Math.round(imgH * MEME_BAR_FRAC) : 0
   276→   276→    const H = imgH + bar * 2
   277→   277→    canvas.width = W
   278→   278→    canvas.height = H
   279→   279→    ctx.clearRect(0, 0, W, H)
   280→   280→    if (bar > 0 || file.type === 'image/jpeg') {
   281→   281→      ctx.fillStyle = '#ffffff'
   282→   282→      ctx.fillRect(0, 0, W, H)
   283→   283→    }
   284→   284→    ctx.drawImage(baseImg, 0, bar)
   285→   285→
   286→   286→    const layouts: Record<string, ElementLayout> = {}
   287→   287→    const metrics: Record<string, number> = {}
   288→   288→    for (const el of state.elements) {
   289→   289→      const lay = paintMemeElement(ctx, W, H, el)
   290→   290→      if (lay) {
   291→   291→        layouts[el.id] = lay
   292→   292→        metrics[el.id] = lay.px
   293→   293→      }
   294→   294→    }
   295→   295→    layoutsRef.current = layouts
   296→   296→
   297→   297→    /* Selection outline for the active element (preview only). */
   298→   298→    const active = activeElementId ? layouts[activeElementId] : null
   299→   299→    if (active) {
   300→   300→      ctx.save()
   301→   301→      ctx.setLineDash([6, 4])
   302→   302→      ctx.lineWidth = Math.max(2, Math.round(W / 400))
   303→   303→      ctx.strokeStyle = 'rgba(139, 92, 246, 0.95)'
   304→   304→      ctx.strokeRect(
   305→   305→        active.bbox.x - 6,
   306→   306→        active.bbox.y - 6,
   307→   307→        active.bbox.w + 12,
   308→   308→        active.bbox.h + 12
   309→   309→      )
   310→   310→      ctx.restore()
   311→   311→    }
   312→   312→    onMetrics(metrics)
   313→   313→  }, [baseImg, state, activeElementId, file, onMetrics])
   314→   314→
   315→   315→  const toCanvas = (e: React.PointerEvent<HTMLCanvasElement>) => {
   316→   316→    const canvas = canvasRef.current!
   317→   317→    const rect = canvas.getBoundingClientRect()
   318→   318→    return {
   319→   319→      cx: ((e.clientX - rect.left) * canvas.width) / rect.width,
   320→   320→      cy: ((e.clientY - rect.top) * canvas.height) / rect.height,
   321→   321→    }
   322→   322→  }
   323→   323→
   324→   324→  const hitTest = (cx: number, cy: number) => {
   325→   325→    const pad = 10
   326→   326→    const entries = state.elements.filter((el) => layoutsRef.current[el.id])
   327→   327→    for (let i = entries.length - 1; i >= 0; i--) {
   328→   328→      const el = entries[i]
   329→   329→      const lay = layoutsRef.current[el.id]
   330→   330→      const inBox =
   331→   331→        cx >= lay.bbox.x - pad &&
   332→   332→        cx <= lay.bbox.x + lay.bbox.w + pad &&
   333→   333→        cy >= lay.bbox.y - pad &&
   334→   334→        cy <= lay.bbox.y + lay.bbox.h + pad
   335→   335→      if (inBox) return el
   336→   336→    }
   337→   337→    return null
   338→   338→  }
   339→   339→
   340→   340→  return (
   341→   341→    <div className="overflow-hidden rounded-2xl border border-border/60" style={CHECKER_BG}>
   342→   342→      {baseImg ? (
   343→   343→        <canvas
   344→   344→          ref={canvasRef}
   345→   345→          className="mx-auto block h-auto max-h-[560px] w-auto max-w-full cursor-move touch-none select-none"
   346→   346→          onPointerDown={(e) => {
   347→   347→            const { cx, cy } = toCanvas(e)
   348→   348→            const hit = hitTest(cx, cy)
   349→   349→            if (hit) {
   350→   350→              onSelect(hit.id)
   351→   351→              const canvas = canvasRef.current!
   352→   352→              dragRef.current = {
   353→   353→                id: hit.id,
   354→   354→                dx: cx - hit.x * canvas.width,
   355→   355→                dy: cy - hit.y * canvas.height,
   356→   356→              }
   357→   357→              canvas.setPointerCapture(e.pointerId)
   358→   358→            } else {
   359→   359→              onSelect(null)
   360→   360→            }
   361→   361→          }}
   362→   362→          onPointerMove={(e) => {
   363→   363→            const drag = dragRef.current
   364→   364→            if (!drag) return
   365→   365→            const { cx, cy } = toCanvas(e)
   366→   366→            const canvas = canvasRef.current!
   367→   367→            const nx = Math.min(0.98, Math.max(0.02, (cx - drag.dx) / canvas.width))
   368→   368→            const ny = Math.min(0.99, Math.max(0.01, (cy - drag.dy) / canvas.height))
   369→   369→            onMove(drag.id, nx, ny)
   370→   370→          }}
   371→   371→          onPointerUp={() => {
   372→   372→            dragRef.current = null
   373→   373→          }}
   374→   374→          onPointerCancel={() => {
   375→   375→            dragRef.current = null
   376→   376→          }}
   377→   377→        />
   378→   378→      ) : (
   379→   379→        <div className="grid h-64 place-items-center">
   380→   380→          <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   381→   381→        </div>
   382→   382→      )}
   383→   383→    </div>
   384→   384→  )
   385→   385→}

   542→   542→        </div>
   543→   543→      </div>
   544→   544→    </div>

   557→   557→
   558→   558→export function MemeMakerView({
   559→   559→  files,
   560→   560→  onRemove,
   561→   561→  onAddMore,
   562→   562→  onChange,
   563→   563→}: MemeMakerViewProps) {
   564→   564→  const [meta, setMeta] = React.useState<Record<string, ImageMeta>>({})
   565→   565→  const [memes, setMemes] = React.useState<Record<string, MemeState>>({})
   566→   566→  const [activeImageId, setActiveImageId] = React.useState<string | null>(null)
   567→   567→  const [activeElementId, setActiveElementId] = React.useState<string | null>(null)
   568→   568→  const [metrics, setMetrics] = React.useState<Record<string, number>>({})
   569→   569→  const urlsRef = React.useRef<Record<string, string>>({})
   570→   570→
   571→   571→  /* ---------------- Load image metadata (size + object URL) ------------- */
   572→   572→  React.useEffect(() => {
   573→   573→    let cancelled = false
   574→   574→
   575→   575→    setMeta((prev) => {
   576→   576→      const next: Record<string, ImageMeta> = {}
   577→   577→      for (const f of files) {
   578→   578→        if (prev[f.id]) next[f.id] = prev[f.id]
   579→   579→      }
   580→   580→      for (const id of Object.keys(urlsRef.current)) {
   581→   581→        if (!next[id] && urlsRef.current[id]) {
   582→   582→          URL.revokeObjectURL(urlsRef.current[id])
   583→   583→          delete urlsRef.current[id]
   584→   584→        }
   585→   585→      }
   586→   586→      return next
   587→   587→    })
   588→   588→
   589→   589→    const missing = files.filter((f) => !urlsRef.current[f.id])
   590→   590→    for (const f of missing) {
   591→   591→      const url = URL.createObjectURL(f.file)
   592→   592→      urlsRef.current[f.id] = url
   593→   593→      const img = new Image()
   594→   594→      img.onload = () => {
   595→   595→        if (cancelled) return
   596→   596→        setMeta((prev) => ({
   597→   597→          ...prev,
   598→   598→          [f.id]: {
   599→   599→            url,
   600→   600→            width: img.naturalWidth || 1,
   601→   601→            height: img.naturalHeight || 1,
   602→   602→          },
   603→   603→        }))
   604→   604→      }
   605→   605→      img.onerror = () => {
   606→   606→        if (cancelled) return
   607→   607→        setMeta((prev) => ({ ...prev, [f.id]: { url, width: 1, height: 1 } }))
   608→   608→      }
   609→   609→      img.src = url
   610→   610→    }
   611→   611→
   612→   612→    return () => {
   613→   613→      cancelled = true
   614→   614→    }
   615→   615→  }, [files])
   616→   616→
   617→   617→  // Revoke everything on unmount.
   618→   618→  React.useEffect(() => {
   619→   619→    const urls = urlsRef.current
   620→   620→    return () => {
   621→   621→      for (const id of Object.keys(urls)) URL.revokeObjectURL(urls[id])
   622→   622→    }
   623→   623→  }, [])
   624→   624→
   625→   625→  /* ---------------- Initialize state for new files ---------------------- */
   626→   626→  React.useEffect(() => {
   627→   627→    setMemes((prev) => {
   628→   628→      let changed = false
   629→   629→      const next = { ...prev }
   630→   630→      for (const f of files) {
   631→   631→        if (!next[f.id]) {
   632→   632→          next[f.id] = makeMemeState()
   633→   633→          changed = true
   634→   634→        }
   635→   635→      }
   636→   636→      for (const id of Object.keys(next)) {
   637→   637→        if (!files.some((f) => f.id === id)) {
   638→   638→          delete next[id]
   639→   639→          changed = true
   640→   640→        }
   641→   641→      }
   642→   642→      return changed ? next : prev
   643→   643→    })
   644→   644→  }, [files])
   645→   645→
   646→   646→  /* Keep the active image valid. */
   647→   647→  React.useEffect(() => {
   648→   648→    if (files.length === 0) {
   649→   649→      setActiveImageId(null)
   650→   650→      return
   651→   651→    }
   652→   652→    setActiveImageId((prev) =>
   653→   653→      prev && files.some((f) => f.id === prev) ? prev : files[0].id
   654→   654→    )
   655→   655→  }, [files])
   656→   656→
   657→   657→  const activeFile = files.find((f) => f.id === activeImageId) ?? files[0] ?? null
   658→   658→  const activeState: MemeState | null = activeFile
   659→   659→    ? memes[activeFile.id] ?? makeMemeState()
   660→   660→    : null
   661→   661→  const activeElement =
   662→   662→    activeState?.elements.find((e) => e.id === activeElementId) ?? null
   663→   663→
   664→   664→  /* Keep the active element valid (per image). */
   665→   665→  React.useEffect(() => {
   666→   666→    if (!activeFile) return
   667→   667→    const st = memes[activeFile.id]
   668→   668→    if (!st) return
   669→   669→    if (activeElementId && st.elements.some((e) => e.id === activeElementId)) return
   670→   670→    setActiveElementId(st.elements.length > 0 ? st.elements[st.elements.length - 1].id : null)
   671→   671→  }, [activeFile, memes, activeElementId])
   672→   672→
   673→   673→  /* ---------------- Emit result whenever memes change -------------------- */
   674→   674→  React.useEffect(() => {
   675→   675→    if (files.length === 0) {
   676→   676→      onChange(null)
   677→   677→      return
   678→   678→    }
   679→   679→    const result: Record<string, MemeState> = {}
   680→   680→    for (const f of files) {
   681→   681→      result[f.file.name] = memes[f.id] ?? makeMemeState()
   682→   682→    }
   683→   683→    onChange({ memes: result })
   684→   684→  }, [files, memes, onChange])
   685→   685→
   686→   686→  /* ---------------- Mutators --------------------------------------------- */
   687→   687→  const patchActive = (patch: Partial<MemeState>) => {
   688→   688→    if (!activeFile) return
   689→   689→    setMemes((prev) => {
   690→   690→      const cur = prev[activeFile.id] ?? makeMemeState()
   691→   691→      return { ...prev, [activeFile.id]: { ...cur, ...patch } }
   692→   692→    })
   693→   693→  }
   694→   694→
   695→   695→  const patchElement = (id: string, patch: Partial<MemeTextElement>) => {
   696→   696→    if (!activeFile) return
   697→   697→    setMemes((prev) => {
   698→   698→      const cur = prev[activeFile.id] ?? makeMemeState()
   699→   699→      return {
   700→   700→        ...prev,
   701→   701→        [activeFile.id]: {
   702→   702→          ...cur,
   703→   703→          elements: cur.elements.map((e) => (e.id === id ? { ...e, ...patch } : e)),
   704→   704→        },
   705→   705→      }
   706→   706→    })
   707→   707→  }
   708→   708→
   709→   709→  const moveElement = (id: string, x: number, y: number) =>
   710→   710→    patchElement(id, { x, y })
   711→   711→
   712→   712→  const addElement = () => {
   713→   713→    if (!activeFile || !activeState) return
   714→   714→    const el = makeElement({ text: 'New text', y: 0.5 })
   715→   715→    patchActive({ elements: [...activeState.elements, el] })
   716→   716→    setActiveElementId(el.id)
   717→   717→  }
   718→   718→
   719→   719→  const duplicateElement = () => {
   720→   720→    if (!activeElement || !activeState) return
   721→   721→    const el = makeElement({
   722→   722→      ...activeElement,
   723→   723→      id: uid('el'),
   724→   724→      y: Math.min(0.95, activeElement.y + 0.08),
   725→   725→    })
   726→   726→    patchActive({ elements: [...activeState.elements, el] })
   727→   727→    setActiveElementId(el.id)
   728→   728→  }
   729→   729→
   730→   730→  const removeElement = (id: string) => {
   731→   731→    if (!activeFile || !activeState) return
   732→   732→    patchActive({ elements: activeState.elements.filter((e) => e.id !== id) })
   733→   733→  }
   734→   734→
   735→   735→  /** Inside ↔ Outside: remap every element's y so text stays glued to the
   736→   736→   *  image content (bars are added/removed around it). */
   737→   737→  const toggleMode = (mode: 'inside' | 'outside') => {
   738→   738→    if (!activeFile || !activeState || activeState.mode === mode) return
   739→   739→    const imgH = (meta[activeFile.id]?.height ?? 1) || 1
   740→   740→    const bar = Math.round(imgH * MEME_BAR_FRAC)
   741→   741→    const total = imgH + bar * 2
   742→   742→    const elements = activeState.elements.map((el) => {
   743→   743→      let y = el.y
   744→   744→      if (mode === 'outside') y = (bar + el.y * imgH) / total
   745→   745→      else y = (el.y * total - bar) / imgH
   746→   746→      return { ...el, y: Math.min(0.99, Math.max(0.01, y)) }
   747→   747→    })
   748→   748→    patchActive({ mode, elements })
   749→   749→  }
   750→   750→
   751→   751→  const elementLabel = (el: MemeTextElement, i: number) =>
   752→   752→    `Text ${i + 1} · ${el.text.trim().replace(/\s+/g, ' ').slice(0, 16) || 'empty'}`
   753→   753→
   754→   754→  const handleMetrics = React.useCallback((m: Record<string, number>) => {
   755→   755→    setMetrics((prev) => {
   756→   756→      const keys = new Set([...Object.keys(prev), ...Object.keys(m)])
   757→   757→      for (const k of keys) if (prev[k] !== m[k]) return m
   758→   758→      return prev
   759→   759→    })
   760→   760→  }, [])
   761→   761→
   762→   762→  const effectivePx = activeElement ? metrics[activeElement.id] ?? 0 : 0
   763→   763→
   764→   764→  return (
   765→   765→    <div className="space-y-4">
   766→   766→      <div className="grid gap-4 lg:grid-cols-[minmax(0,1fr)_360px]">
   767→   767→        {/* ------------------------- Canvas column ------------------------- */}
   768→   768→        <div className="space-y-3">
   769→   769→          {/* Floating text toolbar (iloveimg-style pill, ABOVE the canvas so
   770→   770→              it never blocks dragging the top caption) */}
   771→   771→          {activeElement && (
   772→   772→            <div className="flex justify-center">
   773→   773→              <div className="flex items-center gap-1 rounded-xl border border-border/70 bg-card/95 p-1 shadow-lg backdrop-blur">
   774→   774→                <Select
   775→   775→                  value={activeElement.font}
   776→   776→                  onValueChange={(v) => patchElement(activeElement.id, { font: v })}
   777→   777→                >
   778→   778→                  <SelectTrigger
   779→   779→                    className="h-8 w-[120px] text-xs"
   780→   780→                    aria-label="Font family"
   781→   781→                  >
   782→   782→                    <SelectValue />
   783→   783→                  </SelectTrigger>
   784→   784→                  <SelectContent>
   785→   785→                    {MEME_FONTS.map((f) => (
   786→   786→                      <SelectItem key={f} value={f}>
   787→   787→                        {f}
   788→   788→                      </SelectItem>
   789→   789→                    ))}
   790→   790→                  </SelectContent>
   791→   791→                </Select>
   792→   792→                {(
   793→   793→                  [
   794→   794→                    ['bold', <Bold key="b" className="h-3.5 w-3.5" />, 'Bold'],
   795→   795→                    ['italic', <Italic key="i" className="h-3.5 w-3.5" />, 'Italic'],
   796→   796→                    ['underline', <Underline key="u" className="h-3.5 w-3.5" />, 'Underline'],
   797→   797→                    ['caps', <CaseUpper key="c" className="h-3.5 w-3.5" />, 'ALL-CAPS'],
   798→   798→                  ] as const
   799→   799→                ).map(([key, icon, label]) => (
   800→   800→                  <button
   801→   801→                    key={key}
   802→   802→                    type="button"
   803→   803→                    onClick={() => patchElement(activeElement.id, { [key]: !activeElement[key] })}
   804→   804→                    aria-label={`${label} toggle`}
   805→   805→                    aria-pressed={activeElement[key]}
   806→   806→                    className={cn(
   807→   807→                      'grid h-8 w-8 place-items-center rounded-lg text-muted-foreground transition-colors hover:bg-secondary hover:text-foreground',
   808→   808→                      activeElement[key] && 'bg-primary/10 text-primary hover:bg-primary/10 hover:text-primary'
   809→   809→                    )}
   810→   810→                  >
   811→   811→                    {icon}
   812→   812→                  </button>
   813→   813→                ))}
   814→   814→                <div className="mx-0.5 h-5 w-px bg-border" />
   815→   815→                {(['left', 'center', 'right'] as const).map((a) => {
   816→   816→                  const Icon = a === 'left' ? AlignLeft : a === 'center' ? AlignCenter : AlignRight
   817→   817→                  return (
   818→   818→                    <button
   819→   819→                      key={a}
   820→   820→                      type="button"
   821→   821→                      onClick={() => patchElement(activeElement.id, { align: a })}
   822→   822→                      aria-label={`Align ${a}`}
   823→   823→                      aria-pressed={activeElement.align === a}
   824→   824→                      className={cn(
   825→   825→                        'grid h-8 w-8 place-items-center rounded-lg text-muted-foreground transition-colors hover:bg-secondary hover:text-foreground',
   826→   826→                        activeElement.align === a && 'bg-primary/10 text-primary hover:bg-primary/10 hover:text-primary'
   827→   827→                      )}
   828→   828→                    >
   829→   829→                      <Icon className="h-3.5 w-3.5" />
   830→   830→                    </button>
   831→   831→                  )
   832→   832→                })}
   833→   833→                <div className="mx-0.5 h-5 w-px bg-border" />
   834→   834→                <button
   835→   835→                  type="button"
   836→   836→                  onClick={duplicateElement}
   837→   837→                  aria-label="Duplicate this text"
   838→   838→                  className="grid h-8 w-8 place-items-center rounded-lg text-muted-foreground transition-colors hover:bg-secondary hover:text-foreground"
   839→   839→                >
   840→   840→                  <Copy className="h-3.5 w-3.5" />
   841→   841→                </button>
   842→   842→                <button
   843→   843→                  type="button"
   844→   844→                  onClick={() => removeElement(activeElement.id)}
   845→   845→                  aria-label="Delete this text"
   846→   846→                  className="grid h-8 w-8 place-items-center rounded-lg text-muted-foreground transition-colors hover:bg-destructive/10 hover:text-destructive"
   847→   847→                >
   848→   848→                  <Trash2 className="h-3.5 w-3.5" />
   849→   849→                </button>
   850→   850→              </div>
   851→   851→            </div>
   852→   852→          )}
   853→   853→

   857→   857→              <MemePreview
   858→   858→                file={activeFile.file}
   859→   859→                state={activeState}
   860→   860→                activeElementId={activeElementId}
   861→   861→                onSelect={setActiveElementId}
   862→   862→                onMove={moveElement}
   863→   863→                onMetrics={handleMetrics}
   864→   864→              />
   865→   865→            ) : (
   866→   866→              <div
   867→   867→                className="grid h-64 place-items-center rounded-2xl border border-border/60 text-sm text-muted-foreground"
   868→   868→                style={CHECKER_BG}
   869→   869→              >
   870→   870→                No image selected
   871→   871→              </div>
   872→   872→            )}
   873→   873→
   874→   874→            {/* Floating add-text button */}
   875→   875→            <div className="absolute right-4 top-1/2 z-20 -translate-y-1/2">
   876→   876→              <button
   877→   877→                type="button"
   878→   878→                onClick={addElement}
   879→   879→                aria-label="Add a text box"
   880→   880→                className="relative grid h-12 w-12 place-items-center rounded-full bg-primary text-primary-foreground shadow-lg transition-transform hover:scale-105"
   881→   881→              >
   882→   882→                <Plus className="h-5 w-5" />
   883→   883→                {(activeState?.elements.length ?? 0) > 0 && (
   884→   884→                  <span className="absolute -left-1.5 -top-1.5 grid h-5 min-w-5 place-items-center rounded-full bg-foreground px-1 text-[10px] font-bold text-background">

   888→   888→              </button>
   889→   889→            </div>
   890→   890→          </div>
   891→   891→
   892→   892→          {/* Queued images film strip */}
   893→   893→          <div className="flex gap-2 overflow-x-auto pb-1">
   894→   894→            {files.map((f) => {
   895→   895→              const active = f.id === activeFile?.id
   896→   896→              const url = meta[f.id]?.url
   897→   897→              return (
   898→   898→                <div key={f.id} className="group relative shrink-0">
   899→   899→                  <button
   900→   900→                    type="button"
   901→   901→                    onClick={() => setActiveImageId(f.id)}
   902→   902→                    aria-label={`Edit meme for ${f.file.name}`}
   903→   903→                    aria-pressed={active}
   904→   904→                    className={cn(
   905→   905→                      'block h-14 w-20 overflow-hidden rounded-lg border-2 bg-muted transition-all',
   906→   906→                      active
   907→   907→                        ? 'border-primary'
   908→   908→                        : 'border-transparent opacity-80 hover:opacity-100'
   909→   909→                    )}
   910→   910→                  >
   911→   911→                    {url ? (
   912→   912→                      <img
   913→   913→                        src={url}
   914→   914→                        alt=""
   915→   915→                        className="h-full w-full object-cover"
   916→   916→                        draggable={false}
   917→   917→                      />
   918→   918→                    ) : (
   919→   919→                      <Loader2 className="mx-auto mt-5 h-4 w-4 animate-spin text-muted-foreground" />
   920→   920→                    )}
