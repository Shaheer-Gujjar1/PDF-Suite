   102→   102→/**
   103→   103→ * Resize by moving ONLY the edges/corner the user grabbed.
   104→   104→ *
   105→   105→ * Mid-edge handles (n/s/e/w) are STRICTLY single-axis in every mode — the
   106→   106→ * dragged edge follows the pointer, the opposite edge never moves and the
   107→   107→ * perpendicular dimension is untouched: dragging e/w changes width only,
   108→   108→ * dragging n/s changes height only (iloveimg/Cropper.js behavior). They
   109→   109→ * never consult the ratio: re-centering or re-deriving the other axis made
   110→   110→ * both dimensions change on a side drag and could even snap the width to
   111→   111→ * the container edge when the derived size overflowed.
   112→   112→ *
   113→   113→ * A locked ratio is enforced on CORNER drags only: the dragged corner
   114→   114→ * follows the pointer on the width axis, the height is derived from the
   115→   115→ * ratio and everything is anchored at the opposite corner. Ratio presets
   116→   116→ * still re-fit on click, and Full area / Copy to all / drawing keep their
   117→   117→ * ratio-aware behavior.
   118→   118→ */
   119→   119→function resizeWithHandles(
   120→   120→  dm: DragMode,
   121→   121→  start: PxRect,
   122→   122→  dx: number,
   123→   123→  dy: number,
   124→   124→  cw: number,
   125→   125→  ch: number,
   126→   126→  ratio: number | null
   127→   127→): PxRect {
   128→   128→  let left = start.x
   129→   129→  let top = start.y
   130→   130→  let right = start.x + start.width
   131→   131→  let bottom = start.y + start.height
   132→   132→
   133→   133→  // Which edges did the user grab? Parse the direction from the mode SUFFIX
   134→   134→  // ('n', 'se', …). NEVER dm.includes() on the full string: the word "resize"
   135→   135→  // itself contains 'e' and 's', so 'resize-n'.includes('e') was true — the
   136→   136→  // top handle also dragged the right edge, changing the width on every
   137→   137→  // horizontal swipe.
   138→   138→  const dir = dm.slice('resize-'.length)
   139→   139→  const grabW = dir.includes('w')
   140→   140→  const grabE = dir.includes('e')
   141→   141→  const grabN = dir.includes('n')
   142→   142→  const grabS = dir.includes('s')
   143→   143→  const isCorner = dir.length === 2
   144→   144→
   145→   145→  // 1) Move only the grabbed edges.
   146→   146→  if (grabW) left = start.x + dx
   147→   147→  if (grabE) right = start.x + start.width + dx
   148→   148→  if (grabN) top = start.y + dy
   149→   149→  if (grabS) bottom = start.y + start.height + dy
   150→   150→
   151→   151→  // 2) Min size on each moved edge, anchored to the opposite edge.
   152→   152→  if (grabW && right - left < MIN_SIZE) left = right - MIN_SIZE
   153→   153→  if (grabE && right - left < MIN_SIZE) right = left + MIN_SIZE
   154→   154→  if (grabN && bottom - top < MIN_SIZE) top = bottom - MIN_SIZE
   155→   155→  if (grabS && bottom - top < MIN_SIZE) bottom = top + MIN_SIZE
   156→   156→
   157→   157→  // 3) Keep inside the container without crossing the opposite edge.
   158→   158→  left = clamp(left, 0, cw)
   159→   159→  right = clamp(right, 0, cw)
   160→   160→  top = clamp(top, 0, ch)
   161→   161→  bottom = clamp(bottom, 0, ch)
   162→   162→  if (right - left < MIN_SIZE) {
   163→   163→    if (grabW) left = Math.max(0, right - MIN_SIZE)
   164→   164→    else right = Math.min(cw, left + MIN_SIZE)
   165→   165→  }
   166→   166→  if (bottom - top < MIN_SIZE) {
   167→   167→    if (grabN) top = Math.max(0, bottom - MIN_SIZE)
   168→   168→    else bottom = Math.min(ch, top + MIN_SIZE)
   169→   169→  }
   170→   170→
   171→   171→  // 4) Ratio: corner drags only. Width leads, height follows the ratio and
   172→   172→  // the opposite corner stays anchored. The derived size is capped by the
   173→   173→  // room available FROM THE ANCHOR EDGES so the container clamps can never
   174→   174→  // break the ratio (no off-ratio rects, no size snaps).
   175→   175→  if (ratio && isCorner) {
   176→   176→    const maxW = grabW ? right : cw - left
   177→   177→    const maxH = grabN ? bottom : ch - top
   178→   178→    let w = right - left
   179→   179→    let h = w / ratio
   180→   180→    if (h > maxH) {
   181→   181→      h = maxH
   182→   182→      w = h * ratio
   183→   183→    }
   184→   184→    if (w > maxW) {
   185→   185→      w = maxW
   186→   186→      h = w / ratio
   187→   187→    }
   188→   188→    if (grabW) left = right - w
   189→   189→    else right = left + w
   190→   190→    if (grabN) top = bottom - h
   191→   191→    else bottom = top + h
   192→   192→  }
   193→   193→
   194→   194→  const width = right - left
   195→   195→  const height = bottom - top
   196→   196→  if (width < 2 || height < 2) return start
   197→   197→  return { x: left, y: top, width, height }
   198→   198→}

   200→   200→/** New rect drawn from an anchor point toward the pointer. */
   201→   201→function createRect(
   202→   202→  anchor: PxPoint,
   203→   203→  pos: PxPoint,
   204→   204→  cw: number,
   205→   205→  ch: number,
   206→   206→  ratio: number | null
   207→   207→): PxRect {
   208→   208→  const dirX = pos.x >= anchor.x ? 1 : -1
   209→   209→  const dirY = pos.y >= anchor.y ? 1 : -1
   210→   210→  const maxW = dirX === 1 ? cw - anchor.x : anchor.x
   211→   211→  const maxH = dirY === 1 ? ch - anchor.y : anchor.y
   212→   212→  let w = Math.min(Math.abs(pos.x - anchor.x), maxW)
   213→   213→  let h = Math.min(Math.abs(pos.y - anchor.y), maxH)
   214→   214→  if (ratio) {
   215→   215→    h = w / ratio
   216→   216→    if (h > maxH) {
   217→   217→      h = maxH
   218→   218→      w = h * ratio
   219→   219→    }
   220→   220→  }
   221→   221→  if (w < 2 || h < 2) {
   222→   222→    return { x: anchor.x, y: anchor.y, width: 0, height: 0 }
   223→   223→  }
   224→   224→  return {
   225→   225→    x: dirX === 1 ? anchor.x : anchor.x - w,
   226→   226→    y: dirY === 1 ? anchor.y : anchor.y - h,
   227→   227→    width: w,
   228→   228→    height: h,
   229→   229→  }
   230→   230→}

   232→   232→/** Adjust an existing rect to a ratio, keeping its center. */
   233→   233→function fitRectToRatio(
   234→   234→  rect: PxRect,
   235→   235→  ratio: number,
   236→   236→  cw: number,
   237→   237→  ch: number
   238→   238→): PxRect {
   239→   239→  const cx = rect.x + rect.width / 2
   240→   240→  const cy = rect.y + rect.height / 2
   241→   241→  let w = rect.width
   242→   242→  let h = rect.height
   243→   243→  if (rect.width / rect.height > ratio) h = w / ratio
   244→   244→  else w = h * ratio
   245→   245→  const scale = Math.min(1, cw / w, ch / h)
   246→   246→  w *= scale
   247→   247→  h *= scale
   248→   248→  if (w < MIN_SIZE) {
   249→   249→    w = MIN_SIZE
   250→   250→    h = w / ratio
   251→   251→  }
   252→   252→  if (h < MIN_SIZE) {
   253→   253→    h = MIN_SIZE
   254→   254→    w = h * ratio
   255→   255→  }
   256→   256→  if (w > cw) {
   257→   257→    w = cw
   258→   258→    h = w / ratio
   259→   259→  }
   260→   260→  if (h > ch) {
   261→   261→    h = ch
   262→   262→    w = h * ratio
   263→   263→  }
   264→   264→  return {
   265→   265→    x: clamp(cx - w / 2, 0, Math.max(0, cw - w)),
   266→   266→    y: clamp(cy - h / 2, 0, Math.max(0, ch - h)),
   267→   267→    width: w,
   268→   268→    height: h,
   269→   269→  }
   270→   270→}

   272→   272→/** Centered default selection (used when a preset is clicked with no rect). */
   273→   273→function defaultRect(ratio: number | null, cw: number, ch: number): PxRect {
   274→   274→  const r = ratio ?? cw / ch
   275→   275→  let w = cw * DEFAULT_COVER
   276→   276→  let h = w / r
   277→   277→  if (h > ch * DEFAULT_COVER) {
   278→   278→    h = ch * DEFAULT_COVER
   279→   279→    w = h * r
   280→   280→  }
   281→   281→  return { x: (cw - w) / 2, y: (ch - h) / 2, width: w, height: h }
   282→   282→}

   284→   284→function hitTest(pos: PxPoint, rect: PxRect): DragMode | null {
   285→   285→  const { x, y } = pos
   286→   286→  const left = rect.x
   287→   287→  const top = rect.y
   288→   288→  const right = rect.x + rect.width
   289→   289→  const bottom = rect.y + rect.height
   290→   290→  // Corner zones shrink on small boxes so a mid-edge zone always exists —
   291→   291→  // grabbing the side of a short/narrow box must never register as a corner
   292→   292→  // (a corner would change both dimensions on a one-axis gesture).
   293→   293→  const hz = Math.min(HANDLE_SIZE, rect.width / 2)
   294→   294→  const vz = Math.min(HANDLE_SIZE, rect.height / 2)
   295→   295→
   296→   296→  if (Math.abs(x - left) < hz && Math.abs(y - top) < vz) return 'resize-nw'
   297→   297→  if (Math.abs(x - right) < hz && Math.abs(y - top) < vz) return 'resize-ne'
   298→   298→  if (Math.abs(x - left) < hz && Math.abs(y - bottom) < vz) return 'resize-sw'
   299→   299→  if (Math.abs(x - right) < hz && Math.abs(y - bottom) < vz) return 'resize-se'
   300→   300→  if (Math.abs(y - top) < vz && x > left && x < right) return 'resize-n'
   301→   301→  if (Math.abs(y - bottom) < vz && x > left && x < right) return 'resize-s'
   302→   302→  // Side handles also accept grabs a few px OUTSIDE the edge (the visible
   303→   303→  // border straddles the rect boundary, and a flush edge at x=0 or x=cw
   304→   304→  // previously fell through and started a brand-new selection).
   305→   305→  if (Math.abs(x - left) < hz && y > top && y < bottom) return 'resize-w'
   306→   306→  if (Math.abs(x - right) < hz && y > top && y < bottom) return 'resize-e'
   307→   307→  if (x > left && x < right && y > top && y < bottom) return 'moving'
   308→   308→  return null
   309→   309→}

   323→   323→function formatBytes(bytes: number): string {
   324→   324→  if (bytes === 0) return '0 B'
   325→   325→  const k = 1024
   326→   326→  const sizes = ['B', 'KB', 'MB', 'GB']
   327→   327→  const i = Math.floor(Math.log(bytes) / Math.log(k))
   328→   328→  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`
   329→   329→}
