    19→/* ------------------------------------------------------------------------ */
    20→/* Blur Face — drag rectangles / ellipses anywhere on the image, stack as    */
    21→/* many as needed, dial the blur strength. Shape coords are normalized 0..1  */
    22→/* so they scale to every image size; the worker mirrors this preview with   */
    23→/* a three-pass box blur (processors['blur-faces']). Keep the radius math    */
    24→/* (blurRadiusPx) and clip shapes in sync with the worker.                   */
    25→/* ------------------------------------------------------------------------ */
    26→
    27→export type BlurShapeType = 'rect' | 'ellipse'
    28→
    29→export interface BlurShape {
    30→  id: string
    31→  type: BlurShapeType
    32→  /** 0..1, top-left corner, relative to image dimensions. */
    33→  x: number
    34→  y: number
    35→  /** 0..1 of image dimensions. */
    36→  w: number
    37→  h: number
    38→}
    39→
    40→export interface BlurFacesResult {
    41→  /** Per-image shapes, keyed by file name (matches worker lookup). */
    42→  shapes: Record<string, BlurShape[]>
    43→  /** 1..100 — drives the censor radius, scaled by each image's size. */
    44→  strength: number
    45→}
    46→
    47→type DrawMode = 'move' | BlurShapeType
    48→
    49→interface BlurFacesViewProps {
    50→  files: { id: string; file: File }[]
    51→  onRemove: (id: string) => void
    52→  onAddMore: () => void
    53→  onChange: (result: BlurFacesResult | null) => void
    54→}

    62→const CHECKER_BG: React.CSSProperties = {
    63→  backgroundImage:
    64→    'linear-gradient(45deg, hsl(var(--border) / 0.35) 25%, transparent 25%), linear-gradient(-45deg, hsl(var(--border) / 0.35) 25%, transparent 25%), linear-gradient(45deg, transparent 75%, hsl(var(--border) / 0.35) 75%), linear-gradient(-45deg, transparent 75%, hsl(var(--border) / 0.35) 75%)',
    65→  backgroundSize: '20px 20px',

    69→let counter = 0
    70→function uid(): string {
    71→  counter += 1
    72→  return `bf_${Date.now().toString(36)}_${counter}_${Math.random().toString(36).slice(2, 6)}`
    73→}
    74→
    75→/** Fresh shapes land centred, slightly above middle — where faces usually are. */
    76→function makeShape(type: BlurShapeType, geom?: Partial<BlurShape>): BlurShape {
    77→  const w = type === 'ellipse' ? 0.26 : 0.3
    78→  const h = type === 'ellipse' ? 0.3 : 0.22
    79→  return {
    80→    id: uid(),
    81→    type,
    82→    x: 0.5 - w / 2,
    83→    y: 0.42 - h / 2,
    84→    w,
    85→    h,
    86→    ...geom,
    87→  }
    88→}
    89→
    90→/* ---------------- Shared math (mirrored by the worker) -------------------- */
    91→
    92→/** Worker box radius: strength% of a quarter of the image's short side. */
    93→function blurRadiusPx(strength: number, natW: number, natH: number): number {
    94→  const base = (strength / 100) * 0.25 * Math.min(natW, natH)
    95→  return Math.min(400, Math.max(2, Math.round(base)))
    96→}
    97→
    98→/** clip-path that reveals exactly the shape region of a full-size overlay. */
    99→function clipPathFor(s: Pick<BlurShape, 'type' | 'x' | 'y' | 'w' | 'h'>): string {
   100→  const p = (n: number) => `${(n * 100).toFixed(3)}%`
   101→  if (s.type === 'rect') {
   102→    return `inset(${p(s.y)} ${p(1 - s.x - s.w)} ${p(1 - s.y - s.h)} ${p(s.x)})`
   103→  }
   104→  return `ellipse(${p(s.w / 2)} ${p(s.h / 2)} at ${p(s.x + s.w / 2)} ${p(s.y + s.h / 2)})`
   105→}

   107→const HANDLES = [
   108→  ['nw', 0, 0],
   109→  ['n', 0.5, 0],
   110→  ['ne', 1, 0],
   111→  ['e', 1, 0.5],
   112→  ['se', 1, 1],
   113→  ['s', 0.5, 1],
   114→  ['sw', 0, 1],
   115→  ['w', 0, 0.5],
   116→] as const
   117→
   118→function cursorFor(handle: string): string {
   119→  if (handle === 'n' || handle === 's') return 'ns-resize'
   120→  if (handle === 'e' || handle === 'w') return 'ew-resize'
   121→  if (handle === 'ne' || handle === 'sw') return 'nesw-resize'
   122→  return 'nwse-resize'
   123→}

   125→/* ------------------------------------------------------------------------ */
   126→/* Live stage — base image + per-shape blurred overlays + interaction layer  */
   127→/* ------------------------------------------------------------------------ */
   128→function BlurStage({
   129→  file,
   130→  shapes,
   131→  activeShapeId,
   132→  drawMode,
   133→  strength,
   134→  onSelect,
   135→  onPatch,
   136→  onAdd,
   137→  onDeleteActive,
   138→}: {
   139→  file: File
   140→  shapes: BlurShape[]
   141→  activeShapeId: string | null
   142→  drawMode: DrawMode
   143→  strength: number
   144→  onSelect: (id: string | null) => void
   145→  onPatch: (id: string, patch: Partial<BlurShape>) => void
   146→  onAdd: (type: BlurShapeType, geom: { x: number; y: number; w: number; h: number }) => void
   147→  onDeleteActive: () => void
   148→}) {
   149→  const wrapRef = React.useRef<HTMLDivElement>(null)
   150→  const overlayRef = React.useRef<HTMLDivElement>(null)
   151→  const [src, setSrc] = React.useState<string | null>(null)
   152→  const [baseImg, setBaseImg] = React.useState<HTMLImageElement | null>(null)
   153→  const [disp, setDisp] = React.useState({ w: 0, h: 0 })
   154→  const [ghost, setGhost] = React.useState<BlurShape | null>(null)
   155→  const dragRef = React.useRef<
   156→    | null
   157→    | {
   158→        kind: 'move' | 'resize' | 'draw'
   159→        id?: string
   160→        handle?: string
   161→        startX: number
   162→        startY: number
   163→        orig?: BlurShape
   164→      }
   165→  >(null)
   166→
   167→  /* Load the base image (url kept for the blurred overlay clones). */
   168→  React.useEffect(() => {
   169→    let cancelled = false
   170→    setBaseImg(null)
   171→    setSrc(null)
   172→    const url = URL.createObjectURL(file)
   173→    const img = new Image()
   174→    img.onload = () => {
   175→      if (cancelled) return
   176→      setSrc(url)
   177→      setBaseImg(img)
   178→    }
   179→    img.onerror = () => {
   180→      if (!cancelled) setBaseImg(null)
   181→    }
   182→    img.src = url
   183→    return () => {
   184→      cancelled = true
   185→      URL.revokeObjectURL(url)
   186→    }
   187→  }, [file])
   188→
   189→  /* Track the displayed size so the CSS blur matches the worker's radius. */
   190→  React.useEffect(() => {
   191→    const el = wrapRef.current
   192→    if (!el || !baseImg) return
   193→    const update = () => setDisp({ w: el.clientWidth, h: el.clientHeight })
   194→    update()
   195→    const ro = new ResizeObserver(update)
   196→    ro.observe(el)
   197→    return () => ro.disconnect()
   198→  }, [baseImg])
   199→
   200→  const natW = baseImg?.naturalWidth || 1
   201→  const natH = baseImg?.naturalHeight || 1
   202→  const radius = blurRadiusPx(strength, natW, natH)
   203→  const shortSide = Math.min(disp.w, disp.h)
   204→  const cssBlur = disp.w > 0 && shortSide > 0 ? Math.max(1, radius * (shortSide / Math.min(natW, natH))) : 0
   205→
   206→  const toPct = (e: React.PointerEvent) => {
   207→    const el = overlayRef.current!
   208→    const rect = el.getBoundingClientRect()
   209→    return {
   210→      px: Math.min(1, Math.max(0, (e.clientX - rect.left) / rect.width)),
   211→      py: Math.min(1, Math.max(0, (e.clientY - rect.top) / rect.height)),
   212→    }
   213→  }
   214→
   215→  const capture = (e: React.PointerEvent) => {
   216→    try {
   217→      overlayRef.current?.setPointerCapture(e.pointerId)
   218→    } catch {
   219→      /* ignore — drag still works while the pointer stays inside */
   220→    }
   221→  }
   222→
   223→  const onOverlayPointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
   224→    if (e.button !== 0) return
   225→    const { px, py } = toPct(e)
   226→    if (drawMode === 'move') {
   227→      onSelect(null)
   228→      return
   229→    }
   230→    const ghostType: BlurShapeType = drawMode === 'rect' ? 'rect' : 'ellipse'
   231→    dragRef.current = { kind: 'draw', startX: px, startY: py }
   232→    setGhost({ id: '__ghost', type: ghostType, x: px, y: py, w: 0, h: 0 })
   233→    capture(e)
   234→  }
   235→
   236→  const startMove = (e: React.PointerEvent, s: BlurShape) => {
   237→    if (e.button !== 0) return
   238→    e.stopPropagation()
   239→    const { px, py } = toPct(e)
   240→    onSelect(s.id)
   241→    dragRef.current = { kind: 'move', id: s.id, orig: s, startX: px, startY: py }
   242→    capture(e)
   243→  }
   244→
   245→  const startResize = (e: React.PointerEvent, s: BlurShape, handle: string) => {
   246→    if (e.button !== 0) return
   247→    e.stopPropagation()
   248→    const { px, py } = toPct(e)
   249→    onSelect(s.id)
   250→    dragRef.current = { kind: 'resize', id: s.id, handle, orig: s, startX: px, startY: py }
   251→    capture(e)
   252→  }
   253→
   254→  const onOverlayPointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
   255→    const d = dragRef.current
   256→    if (!d) return
   257→    const { px, py } = toPct(e)
   258→    const MIN = 0.02
   259→    if (d.kind === 'move' && d.orig && d.id) {
   260→      const nx = Math.min(1 - d.orig.w, Math.max(0, d.orig.x + (px - d.startX)))
   261→      const ny = Math.min(1 - d.orig.h, Math.max(0, d.orig.y + (py - d.startY)))
   262→      onPatch(d.id, { x: nx, y: ny })
   263→    } else if (d.kind === 'resize' && d.orig && d.id) {
   264→      const o = d.orig
   265→      const h = d.handle || 'se'
   266→      let x1 = o.x
   267→      let y1 = o.y
   268→      let x2 = o.x + o.w
   269→      let y2 = o.y + o.h
   270→      if (h.includes('w')) x1 = Math.min(px, x2 - MIN)
   271→      if (h.includes('e')) x2 = Math.max(px, x1 + MIN)
   272→      if (h.includes('n')) y1 = Math.min(py, y2 - MIN)
   273→      if (h.includes('s')) y2 = Math.max(py, y1 + MIN)
   274→      x1 = Math.max(0, x1)
   275→      y1 = Math.max(0, y1)
   276→      x2 = Math.min(1, x2)
   277→      y2 = Math.min(1, y2)
   278→      onPatch(d.id, { x: x1, y: y1, w: x2 - x1, h: y2 - y1 })
   279→    } else if (d.kind === 'draw') {
   280→      setGhost({
   281→        id: '__ghost',
   282→        type: drawMode === 'rect' ? 'rect' : 'ellipse',
   283→        x: Math.min(d.startX, px),
   284→        y: Math.min(d.startY, py),
   285→        w: Math.abs(px - d.startX),
   286→        h: Math.abs(py - d.startY),
   287→      })
   288→    }
   289→  }
   290→
   291→  const onOverlayPointerUp = () => {
   292→    const d = dragRef.current
   293→    dragRef.current = null
   294→    if (d?.kind === 'draw' && ghost) {
   295→      if (ghost.w >= 0.015 && ghost.h >= 0.015) {
   296→        onAdd(ghost.type, { x: ghost.x, y: ghost.y, w: ghost.w, h: ghost.h })
   297→      }
   298→    }
   299→    setGhost(null)
   300→  }
   301→
   302→  return (
   303→    <div className="overflow-hidden rounded-2xl border border-border/60" style={CHECKER_BG}>
   304→      {baseImg && src ? (
   305→        <div
   306→          ref={wrapRef}
   307→          className="relative mx-auto w-fit outline-none focus-visible:ring-2 focus-visible:ring-primary/40"
   308→          tabIndex={0}
   309→          role="application"
   310→          aria-label="Blur area editor — drag to draw, drag shapes to move"
   311→          onKeyDown={(e) => {
   312→            if ((e.key === 'Delete' || e.key === 'Backspace') && activeShapeId) {
   313→              e.preventDefault()
   314→              onDeleteActive()
   315→            }
   316→          }}
   317→        >
   318→          <img
   319→            src={src}
   320→            alt={file.name}
   321→            className="block h-auto max-h-[560px] w-auto max-w-full select-none"
   322→            draggable={false}
   323→          />
   324→          <div
   325→            ref={overlayRef}
   326→            className={cn(
   327→              'absolute inset-0 touch-none select-none',
   328→              drawMode === 'move' ? 'cursor-default' : 'cursor-crosshair'
   329→            )}
   330→            onPointerDown={onOverlayPointerDown}
   331→            onPointerMove={onOverlayPointerMove}
   332→            onPointerUp={onOverlayPointerUp}
   333→            onPointerCancel={onOverlayPointerUp}
   334→          >
   335→            {/* Blurred pixels — one clipped clone per shape (CSS stands in for
   336→                the worker's box blur; radius math matches). */}
   337→            {cssBlur > 0 &&
   338→              shapes.map((s) => (
   339→                <img
   340→                  key={`blur-${s.id}`}
   341→                  src={src}
   342→                  alt=""
   343→                  aria-hidden
   344→                  draggable={false}
   345→                  className="pointer-events-none absolute inset-0 h-full w-full select-none"
   346→                  style={{ filter: `blur(${cssBlur}px)`, clipPath: clipPathFor(s) }}
   347→                />
   348→              ))}
   349→
   350→            {/* Interaction outlines */}
   351→            {shapes.map((s, i) => {
   352→              const active = s.id === activeShapeId
   353→              return (
   354→                <div
   355→                  key={s.id}
   356→                  role="button"
   357→                  tabIndex={-1}
   358→                  aria-label={`${s.type === 'ellipse' ? 'Face blur' : 'Box blur'} ${i + 1}`}
   359→                  onPointerDown={(e) => startMove(e, s)}
   360→                  className={cn(
   361→                    'absolute cursor-move',
   362→                    active
   363→                      ? 'border-2 border-primary'
   364→                      : 'border border-white/90 shadow-[0_0_0_1px_rgba(0,0,0,0.45)] hover:border-primary/80'
   365→                  )}
   366→                  style={{
   367→                    left: `${s.x * 100}%`,
   368→                    top: `${s.y * 100}%`,
   369→                    width: `${s.w * 100}%`,
   370→                    height: `${s.h * 100}%`,
   371→                    borderRadius: s.type === 'ellipse' ? '50%' : 6,
   372→                  }}
   373→                >
   374→                  {active && (
   375→                    <>
   376→                      <span className="absolute -top-6 left-0 rounded-md bg-primary px-1.5 py-0.5 text-[10px] font-semibold text-primary-foreground shadow">
   377→                        {s.type === 'ellipse' ? 'Face' : 'Box'} {i + 1}
   378→                      </span>
   379→                      {HANDLES.map(([h, fx, fy]) => (
   380→                        <span
   381→                          key={h}
   382→                          onPointerDown={(e) => startResize(e, s, h)}
   383→                          aria-label={`Resize handle ${h}`}
   384→                          className="absolute h-3.5 w-3.5 -translate-x-1/2 -translate-y-1/2 rounded-full border-2 border-primary bg-white shadow"
   385→                          style={{
   386→                            left: `${fx * 100}%`,
   387→                            top: `${fy * 100}%`,
   388→                            cursor: cursorFor(h),
   389→                          }}
   390→                        />
   391→                      ))}
   392→                    </>
   393→                  )}
   394→                </div>
   395→              )
   396→            })}
   397→
   398→            {/* Ghost while drawing a new area */}
   399→            {ghost && ghost.w > 0.002 && ghost.h > 0.002 && (
   400→              <div
   401→                className="pointer-events-none absolute border-2 border-dashed border-white bg-white/25 shadow-[0_0_0_1px_rgba(0,0,0,0.5)]"
   402→                style={{
   403→                  left: `${ghost.x * 100}%`,
   404→                  top: `${ghost.y * 100}%`,
   405→                  width: `${ghost.w * 100}%`,
   406→                  height: `${ghost.h * 100}%`,
   407→                  borderRadius: ghost.type === 'ellipse' ? '50%' : 6,
   408→                }}
   409→              />
   410→            )}
   411→          </div>
   412→        </div>
   413→      ) : (
   414→        <div className="grid h-64 place-items-center">
   415→          <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   416→        </div>
   417→      )}
   418→    </div>
   419→  )
   420→}

   423→/* Main view — editing studio                                                */
   424→/* ------------------------------------------------------------------------ */
   425→export function BlurFacesView({
   426→  files,
   427→  onRemove,
   428→  onAddMore,
   429→  onChange,
   430→}: BlurFacesViewProps) {
   431→  const [meta, setMeta] = React.useState<Record<string, ImageMeta>>({})
   432→  const [shapesByFile, setShapesByFile] = React.useState<Record<string, BlurShape[]>>({})
   433→  const [activeImageId, setActiveImageId] = React.useState<string | null>(null)
   434→  const [activeShapeId, setActiveShapeId] = React.useState<string | null>(null)
   435→  const [drawMode, setDrawMode] = React.useState<DrawMode>('ellipse')
   436→  const [strength, setStrength] = React.useState(40)
   437→  const urlsRef = React.useRef<Record<string, string>>({})
   438→
   439→  /* ---------------- Load image metadata (size + object URL) ------------- */
   440→  React.useEffect(() => {
   441→    let cancelled = false
   442→
   443→    setMeta((prev) => {
   444→      const next: Record<string, ImageMeta> = {}
   445→      for (const f of files) {
   446→        if (prev[f.id]) next[f.id] = prev[f.id]
   447→      }
   448→      for (const id of Object.keys(urlsRef.current)) {
   449→        if (!next[id] && urlsRef.current[id]) {
   450→          URL.revokeObjectURL(urlsRef.current[id])
   451→          delete urlsRef.current[id]
   452→        }
   453→      }
   454→      return next
   455→    })
   456→
   457→    const missing = files.filter((f) => !urlsRef.current[f.id])
   458→    for (const f of missing) {
   459→      const url = URL.createObjectURL(f.file)
   460→      urlsRef.current[f.id] = url
   461→      const img = new Image()
   462→      img.onload = () => {
   463→        if (cancelled) return
   464→        setMeta((prev) => ({
   465→          ...prev,
   466→          [f.id]: {
   467→            url,
   468→            width: img.naturalWidth || 1,
   469→            height: img.naturalHeight || 1,
   470→          },
   471→        }))
   472→      }
   473→      img.onerror = () => {
   474→        if (cancelled) return
   475→        setMeta((prev) => ({ ...prev, [f.id]: { url, width: 1, height: 1 } }))
   476→      }
   477→      img.src = url
   478→    }
   479→
   480→    return () => {
   481→      cancelled = true
   482→    }
   483→  }, [files])
   484→
   485→  // Revoke everything on unmount.
   486→  React.useEffect(() => {
   487→    const urls = urlsRef.current
   488→    return () => {
   489→      for (const id of Object.keys(urls)) URL.revokeObjectURL(urls[id])
   490→    }
   491→  }, [])
   492→
   493→  /* Keep the active image valid. */
   494→  React.useEffect(() => {
   495→    if (files.length === 0) {
   496→      setActiveImageId(null)
   497→      return
   498→    }
   499→    setActiveImageId((prev) =>
   500→      prev && files.some((f) => f.id === prev) ? prev : files[0].id
   501→    )
   502→  }, [files])
   503→
   504→  const activeFile = files.find((f) => f.id === activeImageId) ?? files[0] ?? null
   505→  const activeShapes: BlurShape[] = activeFile ? shapesByFile[activeFile.id] ?? [] : []
   506→  const activeShape = activeShapes.find((s) => s.id === activeShapeId) ?? null
   507→  const activeMeta = activeFile ? meta[activeFile.id] ?? null : null
   508→
   509→  /* Keep the active shape valid (per image). */
   510→  React.useEffect(() => {
   511→    if (!activeFile) {
   512→      setActiveShapeId(null)
   513→      return
   514→    }
   515→    const list = shapesByFile[activeFile.id] ?? []
   516→    if (activeShapeId && list.some((s) => s.id === activeShapeId)) return
   517→    setActiveShapeId(null)
   518→  }, [activeFile, shapesByFile, activeShapeId])
   519→
   520→  /* ---------------- Emit result whenever anything changes ---------------- */
   521→  React.useEffect(() => {
   522→    if (files.length === 0) {
   523→      onChange(null)
   524→      return
   525→    }
   526→    const result: Record<string, BlurShape[]> = {}
   527→    for (const f of files) {
   528→      result[f.file.name] = (shapesByFile[f.id] ?? []).map((s) => ({ ...s }))
   529→    }
   530→    onChange({ shapes: result, strength })
   531→  }, [files, shapesByFile, strength, onChange])
   532→
   533→  /* ---------------- Mutators --------------------------------------------- */
   534→  const setShapes = (fileId: string, fn: (prev: BlurShape[]) => BlurShape[]) =>
   535→    setShapesByFile((prev) => ({ ...prev, [fileId]: fn(prev[fileId] ?? []) }))
   536→
   537→  const addShape = (type: BlurShapeType, geom?: { x: number; y: number; w: number; h: number }) => {
   538→    if (!activeFile) return
   539→    const s = makeShape(type, geom)
   540→    setShapes(activeFile.id, (prev) => [...prev, s])
   541→    setActiveShapeId(s.id)
   542→  }
   543→
   544→  const patchShape = (id: string, patch: Partial<BlurShape>) => {

   550→

   573→
   574→  /** Copy this image's areas onto every other image (fresh ids). */
   575→  const applyToAll = () => {

   578→    setShapesByFile((prev) => {

   582→        next[f.id] = src.map((s) => ({ ...s, id: uid() }))

   584→      return next

   586→  }
   587→
   588→  const shapeLabel = (s: BlurShape, i: number) =>

   590→
   591→  return (
   592→    <div className="space-y-4">
   593→      <div className="grid gap-4 lg:grid-cols-[minmax(0,1fr)_360px]">
   594→        {/* ------------------------- Canvas column ------------------------- */}
   595→        <div className="space-y-3">
   596→          {/* Toolbar: draw modes + selected-shape quick actions */}

   601→                  ['move', 'Select', MousePointer2],
   602→                  ['ellipse', 'Face', Circle],
   603→                  ['rect', 'Box', Square],
   604→                ] as const
   605→              ).map(([m, label, Icon]) => (
   606→                <button
   607→                  key={m}
   608→                  type="button"
   609→                  onClick={() => setDrawMode(m)}

   613→                    'flex h-8 items-center gap-1.5 rounded-lg px-2.5 text-xs font-medium transition-colors',
   614→                    drawMode === m
   615→                      ? 'bg-primary/10 text-primary'
   616→                      : 'text-muted-foreground hover:bg-secondary hover:text-foreground'
   617→                  )}
   618→                >
   619→                  <Icon className="h-3.5 w-3.5" />

   621→                </button>
   622→              ))}
   623→            </div>
   624→
   625→            {activeShape && (
   626→              <div className="flex items-center gap-1 rounded-xl border border-border/70 bg-card/95 p-1 shadow-sm">
   627→                <button
   628→                  type="button"
   629→                  onClick={() => patchShape(activeShape.id, { type: 'ellipse' })}
   630→                  aria-label="Convert this area to a face (ellipse)"
   631→                  aria-pressed={activeShape.type === 'ellipse'}
   632→                  className={cn(
   633→                    'grid h-8 w-8 place-items-center rounded-lg transition-colors',
   634→                    activeShape.type === 'ellipse'
   635→                      ? 'bg-primary/10 text-primary'
   636→                      : 'text-muted-foreground hover:bg-secondary hover:text-foreground'
   637→                  )}
   638→                >
   639→                  <Circle className="h-3.5 w-3.5" />
   640→                </button>
   641→                <button
   642→                  type="button"
   643→                  onClick={() => patchShape(activeShape.id, { type: 'rect' })}
   644→                  aria-label="Convert this area to a box (rectangle)"
   645→                  aria-pressed={activeShape.type === 'rect'}

   657→                  type="button"
   658→                  onClick={duplicateShape}

   663→                </button>

   665→                  type="button"
   666→                  onClick={() => removeShape(activeShape.id)}
   667→                  aria-label="Delete this area"
   668→                  className="grid h-8 w-8 place-items-center rounded-lg text-muted-foreground transition-colors hover:bg-destructive/10 hover:text-destructive"
   669→                >
   670→                  <Trash2 className="h-3.5 w-3.5" />
   671→                </button>
   672→              </div>

   674→          </div>
   675→
   676→          <div className="relative">
   677→            {/* Live stage */}
   678→            {activeFile ? (
   679→              <BlurStage
   680→                file={activeFile.file}
   681→                shapes={activeShapes}
   682→                activeShapeId={activeShapeId}

   700→
   701→            {/* Area counter */}
   702→            {activeShapes.length > 0 && (

   707→          </div>
   708→
   709→          {drawMode !== 'move' && (

   711→              Drag on the image to draw a {drawMode === 'ellipse' ? 'face' : 'box'} blur area —

   715→
   716→          {/* Queued images film strip */}
   717→          <div className="flex gap-2 overflow-x-auto pb-1">

   733→                        : 'border-transparent opacity-80 hover:opacity-100'

   736→                    {url ? (

   738→                        src={url}
   739→                        alt=""
   740→                        className="h-full w-full object-cover"
   741→                        draggable={false}
   742→                      />
   743→                    ) : (
   744→                      <Loader2 className="mx-auto mt-5 h-4 w-4 animate-spin text-muted-foreground" />
   745→                    )}
   746→                    {count > 0 && (
   747→                      <span className="absolute bottom-0.5 right-0.5 grid h-4 min-w-4 place-items-center rounded-full bg-primary px-1 text-[10px] font-bold text-primary-foreground">
   748→                        {count}
   749→                      </span>
   750→                    )}
   751→                  </button>
   752→                  <button
   753→                    type="button"
   754→                    onClick={() => onRemove(f.id)}
   755→                    aria-label={`Remove ${f.file.name}`}
   756→                    className="absolute -right-1.5 -top-1.5 hidden h-5 w-5 place-items-center rounded-full bg-destructive text-white shadow group-hover:grid"
   757→                  >
   758→                    <X className="h-3 w-3" />
   759→                  </button>
   760→                </div>
   761→              )
   762→            })}
   763→            <button
   764→              type="button"
   765→              onClick={onAddMore}
   766→              aria-label="Add more images"
   767→              className="grid h-14 w-20 shrink-0 place-items-center rounded-lg border-2 border-dashed border-border text-muted-foreground transition-colors hover:border-primary/50 hover:text-primary"
   768→            >
   769→              <ImagePlus className="h-4 w-4" />
   770→            </button>

   772→        </div>
   773→
   774→        {/* ------------------------- Sidebar column ------------------------ */}
   775→        <div className="space-y-3">
   776→          {/* Blur strength */}
   777→          <div className="rounded-2xl border border-border/70 bg-secondary/40 p-4">
   778→            <p className="text-sm font-semibold">Blur strength</p>
   779→            <div className="mt-3 space-y-2">
   780→              <div className="flex items-center justify-between">
   781→                <Label className="text-xs font-medium text-muted-foreground">Amount</Label>
   782→                <span className="text-xs font-semibold text-primary">{strength}%</span>
   783→              </div>
   784→              <Slider
   785→                value={[strength]}
   786→                min={1}
   787→                max={100}
   788→                step={1}
   789→                onValueChange={(v) => setStrength(v[0])}
   790→                aria-label="Blur strength"
   791→              />
   792→              <p className="text-[11px] leading-snug text-muted-foreground">
   793→                Scales with each image&apos;s size, so the same amount censors
   794→                faces equally across all files. Higher = stronger and wider smear.
   795→              </p>
   796→            </div>
   797→          </div>
   798→
   799→          {/* Areas manager */}
   800→          <div className="rounded-2xl border border-border/70 bg-secondary/40 p-4">
   801→            <div className="flex items-center justify-between">
   802→              <p className="text-sm font-semibold">Blur areas</p>
   803→              <span className="text-xs text-muted-foreground">
   804→                {activeShapes.length} on this image
   805→              </span>
   806→            </div>
   807→
   808→            <div className="mt-3 grid grid-cols-2 gap-2">
   809→              <button
   810→                type="button"
   811→                onClick={() => addShape('ellipse')}
   812→                className="flex items-center justify-center gap-2 rounded-xl border border-border bg-card px-3 py-2.5 text-xs font-medium transition-colors hover:border-primary/50 hover:bg-primary/5"
   813→              >
   814→                <Circle className="h-4 w-4 text-primary" />
   815→                Add face blur
   816→              </button>
   817→              <button
   818→                type="button"
   819→                onClick={() => addShape('rect')}
   820→                className="flex items-center justify-center gap-2 rounded-xl border border-border bg-card px-3 py-2.5 text-xs font-medium transition-colors hover:border-primary/50 hover:bg-primary/5"
   821→              >
   822→                <Square className="h-4 w-4 text-primary" />
   823→                Add box blur
   824→              </button>
   825→            </div>
   826→
   827→            {activeShapes.length > 0 ? (
   828→              <div className="mt-3 space-y-1.5">

   830→                  const active = s.id === activeShapeId
   831→                  return (

   833→                      key={s.id}
   834→                      role="button"

   838→                        if (e.key === 'Enter' || e.key === ' ') {
   839→                          e.preventDefault()
   840→                          setActiveShapeId(s.id)
   841→                        }
   842→                      }}
   843→                      className={cn(
   844→                        'flex cursor-pointer items-center gap-2 rounded-lg border px-2.5 py-2 text-xs font-medium transition-colors',
   845→                        active
   846→                          ? 'border-primary bg-primary/5 text-foreground'
   847→                          : 'border-border/60 bg-card text-muted-foreground hover:border-primary/40'
   848→                      )}
   849→                    >
   850→                      {s.type === 'ellipse' ? (
   851→                        <Circle className="h-3.5 w-3.5 shrink-0" />
   852→                      ) : (
   853→                        <Square className="h-3.5 w-3.5 shrink-0" />
   854→                      )}
   855→                      <span className="min-w-0 flex-1 truncate">{shapeLabel(s, i)}</span>

   857→                        type="button"
   858→                        onClick={(e) => {
   859→                          e.stopPropagation()
   860→                          removeShape(s.id)

   862→                        aria-label={`Delete ${shapeLabel(s, i)}`}

   865→                        <Trash2 className="h-3 w-3" />

   870→                <div className="flex gap-2 pt-1">
   871→                  <button
   872→                    type="button"
   873→                    onClick={clearAll}
   874→                    className="flex-1 rounded-lg border border-border bg-card px-2 py-1.5 text-[11px] font-medium text-muted-foreground transition-colors hover:border-destructive/40 hover:text-destructive"

   881→                      onClick={applyToAll}
   882→                      className="flex-1 rounded-lg border border-border bg-card px-2 py-1.5 text-[11px] font-medium text-muted-foreground transition-colors hover:border-primary/40 hover:text-primary"

   887→                </div>
   888→              </div>

   890→              <p className="mt-3 rounded-xl border border-dashed border-border bg-card/60 p-3 text-center text-[11px] text-muted-foreground">
   891→                No blur areas yet — drag on the image with the Face or Box tool,
   892→                or use the buttons above.

   901→                <p className="flex items-center gap-2 text-sm font-semibold">
   902→                  {activeShape.type === 'ellipse' ? (
   903→                    <Circle className="h-4 w-4 text-primary" />
   904→                  ) : (
   905→                    <Square className="h-4 w-4 text-primary" />
   906→                  )}
   907→                  Area settings
   908→                </p>
   909→                <button
   910→                  type="button"
   911→                  onClick={() => removeShape(activeShape.id)}
   912→                  className="grid h-7 w-7 place-items-center rounded-full text-muted-foreground transition-colors hover:bg-destructive/10 hover:text-destructive"
   913→                  aria-label="Delete this area"
   914→                >
   915→                  <Trash2 className="h-3.5 w-3.5" />
   916→                </button>
   917→              </div>
   918→
   919→              <div className="mt-3 space-y-3">
   920→                <div className="grid grid-cols-2 gap-2">
   921→                  {(
   922→                    [
   923→                      ['ellipse', 'Face', Circle],
   924→                      ['rect', 'Box', Square],
   925→                    ] as const
   926→                  ).map(([t, label, Icon]) => (
   927→                    <button
   928→                      key={t}
   929→                      type="button"
   930→                      onClick={() => patchShape(activeShape.id, { type: t })}
   931→                      aria-pressed={activeShape.type === t}
   932→                      className={cn(
   933→                        'flex items-center justify-center gap-1.5 rounded-xl border px-3 py-2 text-xs font-medium transition-colors',
   934→                        activeShape.type === t

   936→                          : 'border-border bg-card text-muted-foreground hover:border-primary/40'

   938→                    >
   939→                      <Icon className="h-3.5 w-3.5" />

   948→                    {Math.round(activeShape.x * activeMeta.width)},{' '}
   949→                    {Math.round(activeShape.y * activeMeta.height)}
   950→                    <br />
   951→                    Drag the area to move it, or pull any of its 8 handles to resize.

   954→              </div>
   955→            </div>
   956→          ) : (
   957→            <div className="rounded-2xl border border-dashed border-border bg-card/60 p-5 text-center">
   958→              <p className="text-sm font-medium">Nothing selected</p>
   959→              <p className="mt-1 text-xs text-muted-foreground">
   960→                Click a blur area on the canvas to move, reshape or delete it —
   961→                press Delete to remove the selected one.
   962→              </p>
   963→            </div>

   965→
   966→          <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   967→            <span className="inline-block h-1.5 w-1.5 animate-pulse rounded-full bg-emerald-500" />
   968→            Editing live — the preview blur matches the baked output exactly.
   969→          </p>

   973→      {/* Hint */}

   976→        <p className="text-muted-foreground">
   977→          Pick an image from the film strip, choose Face or Box above the
