   143→function formatBytes(bytes: number): string {
   144→  if (bytes === 0) return '0 B'
   145→  const k = 1024
   146→  const sizes = ['B', 'KB', 'MB', 'GB']
   147→  const i = Math.floor(Math.log(bytes) / Math.log(k))
   148→  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`
   149→}

   156→function WatermarkPreview({
   157→  file,
   158→  layers,
   159→}: {
   160→  file: File
   161→  layers: WatermarkLayerState[]
   162→}) {
   163→  const canvasRef = React.useRef<HTMLCanvasElement>(null)
   164→  const [baseImg, setBaseImg] = React.useState<HTMLImageElement | null>(null)
   165→  const [logoImgs, setLogoImgs] = React.useState<Record<string, HTMLImageElement>>({})
   166→
   167→  /* Ref mirror so the logo effect below doesn't need `layers` as a dep —
   168→     otherwise every keystroke would revoke + recreate the logo object URLs
   169→     and make the logo flicker while typing. */
   170→  const layersRef = React.useRef(layers)
   171→  React.useEffect(() => {
   172→    layersRef.current = layers
   173→  }, [layers])
   174→
   175→  React.useEffect(() => {
   176→    const url = URL.createObjectURL(file)
   177→    const img = new Image()
   178→    img.onload = () => setBaseImg(img)
   179→    img.onerror = () => setBaseImg(null)
   180→    img.src = url
   181→    return () => {
   182→      URL.revokeObjectURL(url)
   183→      setBaseImg(null)
   184→    }
   185→  }, [file])
   186→
   187→  /* Reload logos ONLY when the actual set of logo files changes. */
   188→  const logoKey = layers
   189→    .map((l) => (l.type === 'image' && l.logo ? `${l.id}:${l.logo.name}:${l.logo.size}` : ''))
   190→    .join('|')
   191→
   192→  React.useEffect(() => {
   193→    const current = layersRef.current
   194→    setLogoImgs({})
   195→    if (current.every((l) => l.type !== 'image' || !l.logo)) return
   196→    let alive = true
   197→    const urls: string[] = []
   198→    for (const l of current) {
   199→      if (l.type !== 'image' || !l.logo) continue
   200→      const url = URL.createObjectURL(l.logo)
   201→      urls.push(url)
   202→      const img = new Image()
   203→      img.onload = () => {
   204→        if (alive) setLogoImgs((prev) => ({ ...prev, [l.id]: img }))
   205→      }
   206→      img.src = url
   207→    }
   208→    return () => {
   209→      alive = false
   210→      for (const u of urls) URL.revokeObjectURL(u)
   211→    }
   212→  }, [logoKey])
   213→
   214→  React.useEffect(() => {
   215→    const canvas = canvasRef.current
   216→    if (!canvas || !baseImg) return
   217→    const ctx = canvas.getContext('2d')
   218→    if (!ctx) return
   219→    const W = baseImg.naturalWidth || 1
   220→    const H = baseImg.naturalHeight || 1
   221→    canvas.width = W
   222→    canvas.height = H
   223→    ctx.clearRect(0, 0, W, H)
   224→    if (file.type === 'image/jpeg') {
   225→      ctx.fillStyle = '#ffffff'
   226→      ctx.fillRect(0, 0, W, H)
   227→    }
   228→
   229→    const drawLayer = (L: WatermarkLayerState) => {
   230→      let tw: number
   231→      let th: number
   232→      if (L.type === 'text') {
   233→        const text = L.text || ''
   234→        if (!text) return
   235→        const size =
   236→          L.fontSizePx > 0
   237→            ? L.fontSizePx
   238→            : Math.max(12, Math.round(Math.min(W, H) / 8))
   239→        ctx.font = `${size}px ${L.fontFamily || 'sans-serif'}`
   240→        ctx.textAlign = 'center'
   241→        ctx.textBaseline = 'middle'
   242→        tw = ctx.measureText(text).width
   243→        th = size
   244→      } else {
   245→        const logo = logoImgs[L.id]
   246→        if (!logo) return
   247→        const scale = Math.max(1, Math.min(100, L.scale || 25)) / 100
   248→        tw = Math.max(1, Math.round(W * scale))
   249→        th = Math.max(1, Math.round(logo.naturalHeight * (tw / logo.naturalWidth)))
   250→      }
   251→      const rad = ((L.rotation || 0) * Math.PI) / 180
   252→      const cos = Math.abs(Math.cos(rad))
   253→      const sin = Math.abs(Math.sin(rad))
   254→      const bboxW = tw * cos + th * sin
   255→      const bboxH = tw * sin + th * cos
   256→      const mX = Math.max(0, L.marginX || 0)
   257→      const mY = Math.max(0, L.marginY || 0)
   258→      const stamp = (cx: number, cy: number) => {
   259→        ctx.save()
   260→        ctx.globalAlpha = Math.max(0, Math.min(1, L.opacity == null ? 0.5 : L.opacity))
   261→        ctx.translate(cx, cy)
   262→        ctx.rotate(rad)
   263→        if (L.type === 'text') {
   264→          ctx.fillStyle = L.color || '#000000'
   265→          ctx.fillText(L.text, 0, 0)
   266→        } else {
   267→          ctx.drawImage(logoImgs[L.id], -tw / 2, -th / 2, tw, th)
   268→        }
   269→        ctx.restore()
   270→      }
   271→      if (L.tile) {
   272→        const stepX = bboxW + mX * 2
   273→        const stepY = bboxH + mY * 2
   274→        for (let yy = -stepY; yy < H + stepY; yy += stepY) {
   275→          for (let xx = -stepX; xx < W + stepX; xx += stepX) {
   276→            stamp(xx + stepX / 2, yy + stepY / 2)
   277→          }
   278→        }
   279→      } else {
   280→        const pos = L.position || 'bottom-right'
   281→        const cx = pos.includes('left')
   282→          ? mX + bboxW / 2
   283→          : pos.includes('right')
   284→            ? W - mX - bboxW / 2
   285→            : W / 2
   286→        const cy = pos.includes('top')
   287→          ? mY + bboxH / 2
   288→          : pos.includes('bottom')
   289→            ? H - mY - bboxH / 2
   290→            : H / 2
   291→        stamp(cx, cy)
   292→      }
   293→    }
   294→
   295→    for (const l of layers.filter((x) => x.over === false)) drawLayer(l)
   296→    ctx.globalAlpha = 1
   297→    ctx.drawImage(baseImg, 0, 0)
   298→    for (const l of layers.filter((x) => x.over !== false)) drawLayer(l)
   299→    ctx.globalAlpha = 1
   300→  }, [baseImg, logoImgs, layers, file])
   301→
   302→  return (
   303→    <div
   304→      className="overflow-hidden rounded-2xl border border-border/60"
   305→      style={CHECKER_BG}
   306→    >
   307→      {baseImg ? (
   308→        <canvas
   309→          ref={canvasRef}
   310→          className="mx-auto block h-auto max-h-[560px] w-auto max-w-full"
   311→        />
   312→      ) : (
   313→        <div className="grid h-64 place-items-center">
   314→          <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   315→        </div>
   316→      )}
   317→    </div>
   318→  )
   319→}

   324→function LayerCard({
   325→  layer,
   326→  index,
   327→  onChange,
   328→  onRemove,
   329→}: {
   330→  layer: WatermarkLayerState
   331→  index: number
   332→  onChange: (patch: Partial<WatermarkLayerState>) => void
   333→  onRemove: () => void
   334→}) {
   335→  const isText = layer.type === 'text'
   336→  return (
   337→    <div className="rounded-2xl border border-border/70 bg-card p-4">
   338→      {/* Header */}
   339→      <div className="mb-4 flex items-center gap-2">
   340→        <span className="grid h-7 w-7 place-items-center rounded-lg bg-primary/10 text-primary">
   341→          {isText ? <TypeIcon className="h-4 w-4" /> : <StampIcon className="h-4 w-4" />}
   342→        </span>
   343→        <p className="flex-1 text-sm font-medium">
   344→          {isText ? 'Text watermark' : 'Logo watermark'}
   345→          <span className="ml-2 text-xs font-normal text-muted-foreground">
   346→            layer {index + 1}
   347→          </span>
   348→        </p>
   349→        <button
   350→          type="button"
   351→          onClick={onRemove}
   352→          className="grid h-7 w-7 place-items-center rounded-full text-muted-foreground transition-colors hover:bg-destructive/10 hover:text-destructive"
   353→          aria-label="Remove this watermark layer"

   355→          <Trash2 className="h-3.5 w-3.5" />
   356→        </button>
   357→      </div>
   358→
   359→      {/* Type-specific controls */}
   360→      {isText ? (
   361→        <div className="space-y-4">

   373→              <Label className="text-xs font-medium text-muted-foreground">Font</Label>
   374→              <Select
   375→                value={layer.fontFamily}
   376→                onValueChange={(v) => onChange({ fontFamily: v })}
   377→              >
   378→                <SelectTrigger aria-label="Watermark font">
   379→                  <SelectValue />
   380→                </SelectTrigger>
   381→                <SelectContent>
   382→                  {FONTS.map((f) => (
   383→                    <SelectItem key={f.value} value={f.value}>
   384→                      {f.label}
   385→                    </SelectItem>
   386→                  ))}
   387→                </SelectContent>
   388→              </Select>
   389→            </div>
   390→            <div className="space-y-2">
   391→              <Label className="text-xs font-medium text-muted-foreground">Color</Label>
   392→              <input
   393→                type="color"
   394→                value={layer.color}
   395→                onChange={(e) => onChange({ color: e.target.value })}
   396→                aria-label="Watermark color"
   397→                className="h-9 w-full cursor-pointer rounded-md border border-border bg-transparent p-1"
   398→              />
   399→            </div>
   400→          </div>
   401→          <div className="space-y-2">
   402→            <div className="flex items-center justify-between">
   403→              <Label className="text-xs font-medium text-muted-foreground">
   404→                Size {layer.fontSizePx > 0 ? `· ${layer.fontSizePx} px` : '· auto-fit'}
   405→              </Label>
   406→              <span className="flex items-center gap-2 text-xs text-muted-foreground">
   407→                Auto
   408→                <Switch
   409→                  checked={layer.fontSizePx === 0}
   410→                  onCheckedChange={(v) => onChange({ fontSizePx: v ? 0 : 48 })}
   411→                  aria-label="Automatic text size"
   412→                />
   413→              </span>
   414→            </div>
   415→            <Slider
   416→              value={[layer.fontSizePx > 0 ? layer.fontSizePx : 48]}
   417→              min={8}
   418→              max={200}
   419→              step={1}
   420→              disabled={layer.fontSizePx === 0}
   421→              onValueChange={(v) => onChange({ fontSizePx: v[0] })}
   422→              aria-label="Text size in pixels"
   423→            />
   424→          </div>
   425→        </div>
   426→      ) : (

   433→                  src={layer.logoUrl}

   445→                  type="file"

   619→export function WatermarkImagesView({

   622→  onAddMore,
   623→  onChange,
   624→}: WatermarkImagesViewProps) {
   625→  const [meta, setMeta] = React.useState<Record<string, ImageMeta>>({})
   626→  const [layers, setLayers] = React.useState<WatermarkLayerState[]>([])
   627→  const [activeLayerId, setActiveLayerId] = React.useState<string | null>(null)
   628→  const [activeImageId, setActiveImageId] = React.useState<string | null>(null)
   629→  const [addOpen, setAddOpen] = React.useState(false)
   630→  const urlsRef = React.useRef<Record<string, string>>({})
   631→
   632→  /* Revoke logo object URLs on unmount. */
   633→  const layersRef = React.useRef<WatermarkLayerState[]>([])
   634→  React.useEffect(() => {
   635→    layersRef.current = layers
   636→  }, [layers])
   637→  React.useEffect(
   638→    () => () => {
   639→      for (const l of layersRef.current) {
   640→        if (l.logoUrl) URL.revokeObjectURL(l.logoUrl)
   641→      }
   642→    },
   643→    []
   644→  )
   645→
   646→  /* ---------------- Load image metadata (size + object URL) ------------- */
   647→  React.useEffect(() => {
   648→    let cancelled = false
   649→
   650→    setMeta((prev) => {
   651→      const next: Record<string, ImageMeta> = {}
   652→      for (const f of files) {
   653→        if (prev[f.id]) next[f.id] = prev[f.id]
   654→      }
   655→      for (const id of Object.keys(urlsRef.current)) {
   656→        if (!next[id] && urlsRef.current[id]) {
   657→          URL.revokeObjectURL(urlsRef.current[id])
   658→          delete urlsRef.current[id]
   659→        }
   660→      }
   661→      return next
   662→    })
   663→
   664→    const missing = files.filter((f) => !urlsRef.current[f.id])
   665→    for (const f of missing) {
   666→      const url = URL.createObjectURL(f.file)
   667→      urlsRef.current[f.id] = url
   668→      const img = new Image()
   669→      img.onload = () => {
   670→        if (cancelled) return
   671→        setMeta((prev) => ({
   672→          ...prev,
   673→          [f.id]: {
   674→            url,
   675→            width: img.naturalWidth || 1,
   676→            height: img.naturalHeight || 1,
   677→          },
   678→        }))
   679→      }
   680→      img.onerror = () => {
   681→        if (cancelled) return
   682→        setMeta((prev) => ({ ...prev, [f.id]: { url, width: 1, height: 1 } }))
   683→      }
   684→      img.src = url
   685→    }
   686→
   687→    return () => {
   688→      cancelled = true
   689→    }
   690→  }, [files])
   691→
   692→  // Revoke everything on unmount.
   693→  React.useEffect(() => {
   694→    const urls = urlsRef.current
   695→    return () => {
   696→      for (const id of Object.keys(urls)) URL.revokeObjectURL(urls[id])
   697→    }
   698→  }, [])
   699→
   700→  /* ---------------- Emit result whenever layers change ------------------- */
   701→  React.useEffect(() => {
   702→    if (files.length === 0 || layers.length === 0) {
   703→      onChange(null)
   704→      return
   705→    }
   706→    onChange({ layers })
   707→  }, [files, layers, onChange])
   708→
   709→  /* Keep the previewed image valid as the queue changes. */
   710→  React.useEffect(() => {
   711→    if (files.length === 0) {
   712→      setActiveImageId(null)
   713→      return
   714→    }
   715→    setActiveImageId((prev) =>
   716→      prev && files.some((f) => f.id === prev) ? prev : files[0].id
   717→    )
   718→  }, [files])
   719→
   720→  /* Keep the active layer valid as layers come and go. */
   721→  React.useEffect(() => {
   722→    if (layers.length === 0) {
   723→      setActiveLayerId(null)
   724→      return
   725→    }
   726→    setActiveLayerId((prev) =>
   727→      prev && layers.some((l) => l.id === prev) ? prev : layers[layers.length - 1].id
   728→    )
   729→  }, [layers])
   730→
   731→  const activeFile = files.find((f) => f.id === activeImageId) ?? files[0] ?? null
   732→  const activeLayer = layers.find((l) => l.id === activeLayerId) ?? null
   733→
   734→  const patchLayer = (id: string, patch: Partial<WatermarkLayerState>) =>
   735→    setLayers((prev) => prev.map((l) => (l.id === id ? { ...l, ...patch } : l)))
   736→
   737→  const removeLayer = (id: string) =>
   738→    setLayers((prev) => {
   739→      const target = prev.find((l) => l.id === id)
   740→      if (target?.logoUrl) URL.revokeObjectURL(target.logoUrl)
   741→      return prev.filter((l) => l.id !== id)

   773→            {layers.length > 0 && (
   774→              <div className="absolute left-1/2 top-3 z-20 -translate-x-1/2">

   776→                  <button
   777→                    type="button"
   778→                    onClick={() => activeLayer && moveLayer(activeLayer.id, 1)}
   779→                    disabled={!activeLayer || layers.indexOf(activeLayer) === layers.length - 1}

   781→                    aria-label="Bring layer forward (moves it on top)"
   782→                  >
   783→                    <ArrowUp className="h-4 w-4" />
   784→                  </button>
   785→                  <button

   802→                      <SelectValue placeholder="Pick a layer…" />
   803→                    </SelectTrigger>
   804→                    <SelectContent>
   805→                      {layers.map((l, i) => (
   806→                        <SelectItem key={l.id} value={l.id}>
   807→                          {layerLabel(l, i)}
   808→                        </SelectItem>
   809→                      ))}
   810→                    </SelectContent>
   811→                  </Select>
   812→                </div>
   813→              </div>
   814→            )}
   815→
   816→            {/* Live canvas */}
   817→            {activeFile ? (
   818→              <WatermarkPreview file={activeFile.file} layers={layers} />

   821→                className="grid h-64 place-items-center rounded-2xl border border-border/60 text-sm text-muted-foreground"
