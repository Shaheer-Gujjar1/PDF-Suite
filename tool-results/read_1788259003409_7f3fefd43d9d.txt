   102→   102→/**
   103→   103→ * Resize by moving ONLY the edges/corner the user grabbed.
   104→   104→ *
   105→   105→ * Mid-edge handles (n/s/e/w) are STRICTLY single-axis in every mode — the
   106→   106→ * dragged edge follows the pointer, the opposite edge never moves and the
   107→   107→ * perpendicular dimension is untouched: dragging e/w changes width only,
   108→   108→ * dragging n/s changes height only (iloveimg/Cropper.js behavior). They
   109→   109→ * never consult the ratio: re-centering or re-deriving the other axis made
   110→   110→ * both dimensions change on a side drag and could even snap the width to
   111→   111→ * the container edge when the derived size overflowed.
   112→   112→ *
   113→   113→ * A locked ratio is enforced on CORNER drags only: the dragged corner
   114→   114→ * follows the pointer on the width axis, the height is derived from the
   115→   115→ * ratio and everything is anchored at the opposite corner. Ratio presets
   116→   116→ * still re-fit on click, and Full area / Copy to all / drawing keep their
   117→   117→ * ratio-aware behavior.
   118→   118→ */
   119→   119→function resizeWithHandles(
   120→   120→  dm: DragMode,
   121→   121→  start: PxRect,
   122→   122→  dx: number,
   123→   123→  dy: number,
   124→   124→  cw: number,
   125→   125→  ch: number,
   126→   126→  ratio: number | null
   127→   127→): PxRect {
   128→   128→  let left = start.x
   129→   129→  let top = start.y
   130→   130→  let right = start.x + start.width
   131→   131→  let bottom = start.y + start.height
   132→   132→
   133→   133→  // Which edges did the user grab? Parse the direction from the mode SUFFIX
   134→   134→  // ('n', 'se', …). NEVER dm.includes() on the full string: the word "resize"
   135→   135→  // itself contains 'e' and 's', so 'resize-n'.includes('e') was true — the
   136→   136→  // top handle also dragged the right edge, changing the width on every
   137→   137→  // horizontal swipe.
   138→   138→  const dir = dm.slice('resize-'.length)
   139→   139→  const grabW = dir.includes('w')
   140→   140→  const grabE = dir.includes('e')
   141→   141→  const grabN = dir.includes('n')
   142→   142→  const grabS = dir.includes('s')
   143→   143→  const isCorner = dir.length === 2
   144→   144→
   145→   145→  // 1) Move only the grabbed edges.
   146→   146→  if (grabW) left = start.x + dx
   147→   147→  if (grabE) right = start.x + start.width + dx
   148→   148→  if (grabN) top = start.y + dy
   149→   149→  if (grabS) bottom = start.y + start.height + dy
   150→   150→
   151→   151→  // 2) Min size on each moved edge, anchored to the opposite edge.
   152→   152→  if (grabW && right - left < MIN_SIZE) left = right - MIN_SIZE
   153→   153→  if (grabE && right - left < MIN_SIZE) right = left + MIN_SIZE
   154→   154→  if (grabN && bottom - top < MIN_SIZE) top = bottom - MIN_SIZE
   155→   155→  if (grabS && bottom - top < MIN_SIZE) bottom = top + MIN_SIZE
   156→   156→
   157→   157→  // 3) Keep inside the container without crossing the opposite edge.
   158→   158→  left = clamp(left, 0, cw)
   159→   159→  right = clamp(right, 0, cw)
   160→   160→  top = clamp(top, 0, ch)
   161→   161→  bottom = clamp(bottom, 0, ch)
   162→   162→  if (right - left < MIN_SIZE) {
   163→   163→    if (grabW) left = Math.max(0, right - MIN_SIZE)
   164→   164→    else right = Math.min(cw, left + MIN_SIZE)
   165→   165→  }
   166→   166→  if (bottom - top < MIN_SIZE) {
   167→   167→    if (grabN) top = Math.max(0, bottom - MIN_SIZE)
   168→   168→    else bottom = Math.min(ch, top + MIN_SIZE)
   169→   169→  }
   170→   170→
   171→   171→  // 4) Ratio: corner drags only. Width leads, height follows the ratio and
   172→   172→  // the opposite corner stays anchored. The derived size is capped by the
   173→   173→  // room available FROM THE ANCHOR EDGES so the container clamps can never
   174→   174→  // break the ratio (no off-ratio rects, no size snaps).
   175→   175→  if (ratio && isCorner) {
   176→   176→    const maxW = grabW ? right : cw - left
   177→   177→    const maxH = grabN ? bottom : ch - top
   178→   178→    let w = right - left
   179→   179→    let h = w / ratio
   180→   180→    if (h > maxH) {
   181→   181→      h = maxH
   182→   182→      w = h * ratio
   183→   183→    }
   184→   184→    if (w > maxW) {
   185→   185→      w = maxW
   186→   186→      h = w / ratio
   187→   187→    }
   188→   188→    if (grabW) left = right - w
   189→   189→    else right = left + w
   190→   190→    if (grabN) top = bottom - h
   191→   191→    else bottom = top + h
   192→   192→  }
   193→   193→
   194→   194→  const width = right - left
   195→   195→  const height = bottom - top
   196→   196→  if (width < 2 || height < 2) return start
   197→   197→  return { x: left, y: top, width, height }
   198→   198→}

   200→   200→/** New rect drawn from an anchor point toward the pointer. */
   201→   201→function createRect(
   202→   202→  anchor: PxPoint,
   203→   203→  pos: PxPoint,
   204→   204→  cw: number,
   205→   205→  ch: number,
   206→   206→  ratio: number | null
   207→   207→): PxRect {
   208→   208→  const dirX = pos.x >= anchor.x ? 1 : -1
   209→   209→  const dirY = pos.y >= anchor.y ? 1 : -1
   210→   210→  const maxW = dirX === 1 ? cw - anchor.x : anchor.x
   211→   211→  const maxH = dirY === 1 ? ch - anchor.y : anchor.y
   212→   212→  let w = Math.min(Math.abs(pos.x - anchor.x), maxW)
   213→   213→  let h = Math.min(Math.abs(pos.y - anchor.y), maxH)
   214→   214→  if (ratio) {
   215→   215→    h = w / ratio
   216→   216→    if (h > maxH) {
   217→   217→      h = maxH
   218→   218→      w = h * ratio
   219→   219→    }
   220→   220→  }
   221→   221→  if (w < 2 || h < 2) {
   222→   222→    return { x: anchor.x, y: anchor.y, width: 0, height: 0 }
   223→   223→  }
   224→   224→  return {
   225→   225→    x: dirX === 1 ? anchor.x : anchor.x - w,
   226→   226→    y: dirY === 1 ? anchor.y : anchor.y - h,
   227→   227→    width: w,
   228→   228→    height: h,
   229→   229→  }
   230→   230→}

   232→   232→/** Adjust an existing rect to a ratio, keeping its center. */
   233→   233→function fitRectToRatio(
   234→   234→  rect: PxRect,
   235→   235→  ratio: number,
   236→   236→  cw: number,
   237→   237→  ch: number
   238→   238→): PxRect {
   239→   239→  const cx = rect.x + rect.width / 2
   240→   240→  const cy = rect.y + rect.height / 2
   241→   241→  let w = rect.width
   242→   242→  let h = rect.height
   243→   243→  if (rect.width / rect.height > ratio) h = w / ratio
   244→   244→  else w = h * ratio
   245→   245→  const scale = Math.min(1, cw / w, ch / h)
   246→   246→  w *= scale
   247→   247→  h *= scale
   248→   248→  if (w < MIN_SIZE) {
   249→   249→    w = MIN_SIZE
   250→   250→    h = w / ratio
   251→   251→  }
   252→   252→  if (h < MIN_SIZE) {
   253→   253→    h = MIN_SIZE
   254→   254→    w = h * ratio
   255→   255→  }
   256→   256→  if (w > cw) {
   257→   257→    w = cw
   258→   258→    h = w / ratio
   259→   259→  }
   260→   260→  if (h > ch) {
   261→   261→    h = ch
   262→   262→    w = h * ratio
   263→   263→  }
   264→   264→  return {
   265→   265→    x: clamp(cx - w / 2, 0, Math.max(0, cw - w)),
   266→   266→    y: clamp(cy - h / 2, 0, Math.max(0, ch - h)),
   267→   267→    width: w,
   268→   268→    height: h,
   269→   269→  }
   270→   270→}

   272→   272→/** Centered default selection (used when a preset is clicked with no rect). */
   273→   273→function defaultRect(ratio: number | null, cw: number, ch: number): PxRect {
   274→   274→  const r = ratio ?? cw / ch
   275→   275→  let w = cw * DEFAULT_COVER
   276→   276→  let h = w / r
   277→   277→  if (h > ch * DEFAULT_COVER) {
   278→   278→    h = ch * DEFAULT_COVER
   279→   279→    w = h * r
   280→   280→  }
   281→   281→  return { x: (cw - w) / 2, y: (ch - h) / 2, width: w, height: h }
   282→   282→}

   284→   284→function hitTest(pos: PxPoint, rect: PxRect): DragMode | null {
   285→   285→  const { x, y } = pos
   286→   286→  const left = rect.x
   287→   287→  const top = rect.y
   288→   288→  const right = rect.x + rect.width
   289→   289→  const bottom = rect.y + rect.height
   290→   290→  // Corner zones shrink on small boxes so a mid-edge zone always exists —
   291→   291→  // grabbing the side of a short/narrow box must never register as a corner
   292→   292→  // (a corner would change both dimensions on a one-axis gesture).
   293→   293→  const hz = Math.min(HANDLE_SIZE, rect.width / 2)
   294→   294→  const vz = Math.min(HANDLE_SIZE, rect.height / 2)
   295→   295→
   296→   296→  if (Math.abs(x - left) < hz && Math.abs(y - top) < vz) return 'resize-nw'
   297→   297→  if (Math.abs(x - right) < hz && Math.abs(y - top) < vz) return 'resize-ne'
   298→   298→  if (Math.abs(x - left) < hz && Math.abs(y - bottom) < vz) return 'resize-sw'
   299→   299→  if (Math.abs(x - right) < hz && Math.abs(y - bottom) < vz) return 'resize-se'
   300→   300→  if (Math.abs(y - top) < vz && x > left && x < right) return 'resize-n'
   301→   301→  if (Math.abs(y - bottom) < vz && x > left && x < right) return 'resize-s'
   302→   302→  // Side handles also accept grabs a few px OUTSIDE the edge (the visible
   303→   303→  // border straddles the rect boundary, and a flush edge at x=0 or x=cw
   304→   304→  // previously fell through and started a brand-new selection).
   305→   305→  if (Math.abs(x - left) < hz && y > top && y < bottom) return 'resize-w'
   306→   306→  if (Math.abs(x - right) < hz && y > top && y < bottom) return 'resize-e'
   307→   307→  if (x > left && x < right && y > top && y < bottom) return 'moving'
   308→   308→  return null
   309→   309→}

   323→   323→function formatBytes(bytes: number): string {
   324→   324→  if (bytes === 0) return '0 B'
   325→   325→  const k = 1024
   326→   326→  const sizes = ['B', 'KB', 'MB', 'GB']
   327→   327→  const i = Math.floor(Math.log(bytes) / Math.log(k))
   328→   328→  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`
   329→   329→}

   331→   331→export function CropImagesView({
   332→   332→  files,
   333→   333→  onRemove,
   334→   334→  onAddMore,
   335→   335→  onChange,
   336→   336→}: CropImagesViewProps) {

   338→   338→  const [activeId, setActiveId] = React.useState<string | null>(null)
   339→   339→  const [relCrops, setRelCrops] = React.useState<Record<string, RelRect>>({})
   340→   340→  const [ratio, setRatio] = React.useState<Ratio>('free')

   345→   345→  // handlers, so mutating it outside render is safe (and lint-clean).
   346→   346→  // `ratio` locks the aspect ratio for the whole drag, `lastRect` keeps the
   347→   347→  // newest px rect so the drag-end check never depends on render timing.
   348→   348→  const drag = React.useRef<{
   349→   349→    mode: DragMode | 'creating'
   350→   350→    anchor: PxPoint
   351→   351→    start: PxRect | null

   353→   353→    ratio: number | null
   354→   354→    lastRect: PxRect | null
   355→   355→  } | null>(null)

   357→   357→  const urlsRef = React.useRef<Record<string, string>>({})
   358→   358→
   359→   359→  /* ---------------- Load image metadata (size + object URL) ------------- */
   360→   360→  React.useEffect(() => {
   361→   361→    let cancelled = false
   362→   362→
   363→   363→    setMeta((prev) => {
   364→   364→      const next: Record<string, ImageMeta> = {}
   365→   365→      for (const f of files) {
   366→   366→        if (prev[f.id]) next[f.id] = prev[f.id]
   367→   367→      }
   368→   368→      for (const id of Object.keys(urlsRef.current)) {
   369→   369→        if (!next[id] && urlsRef.current[id]) {
   370→   370→          URL.revokeObjectURL(urlsRef.current[id])
   371→   371→          delete urlsRef.current[id]
   372→   372→        }
   373→   373→      }
   374→   374→      return next
   375→   375→    })
   376→   376→
   377→   377→    const missing = files.filter((f) => !urlsRef.current[f.id])
   378→   378→    for (const f of missing) {
   379→   379→      const url = URL.createObjectURL(f.file)
   380→   380→      urlsRef.current[f.id] = url
   381→   381→      const img = new Image()
   382→   382→      img.onload = () => {
   383→   383→        if (cancelled) return
   384→   384→        setMeta((prev) => ({
   385→   385→          ...prev,
   386→   386→          [f.id]: {
   387→   387→            url,
   388→   388→            width: img.naturalWidth || 1,
   389→   389→            height: img.naturalHeight || 1,
   390→   390→          },
   391→   391→        }))
   392→   392→      }
   393→   393→      img.onerror = () => {
   394→   394→        if (cancelled) return
   395→   395→        setMeta((prev) => ({
   396→   396→          ...prev,
   397→   397→          [f.id]: { url, width: 1, height: 1 },
   398→   398→        }))
   399→   399→      }
   400→   400→      img.src = url
   401→   401→    }
   402→   402→
   403→   403→    return () => {
   404→   404→      cancelled = true
   405→   405→    }
   406→   406→  }, [files])
   407→   407→
   408→   408→  // Revoke everything on unmount.
   409→   409→  React.useEffect(() => {
   410→   410→    const urls = urlsRef.current
   411→   411→    return () => {
   412→   412→      for (const id of Object.keys(urls)) URL.revokeObjectURL(urls[id])
   413→   413→    }
   414→   414→  }, [])
   415→   415→
   416→   416→  /* ---------------- Keep a valid active selection ------------------------ */
   417→   417→  React.useEffect(() => {
   418→   418→    if (files.length === 0) {
   419→   419→      setActiveId(null)
   420→   420→      return
   421→   421→    }
   422→   422→    if (!activeId || !files.some((f) => f.id === activeId)) {
   423→   423→      const firstWithoutCrop = files.find((f) => !relCrops[f.id]) ?? files[0]
   424→   424→      setActiveId(firstWithoutCrop.id)
   425→   425→    }
   426→   426→  }, [files, activeId, relCrops])
   427→   427→
   428→   428→  /* ---------------- Editing state for the active image ------------------- */
   429→   429→  // NOTE: must be declared before the layout effect below — dependency arrays
   430→   430→  // evaluate synchronously during render, so referencing `activeMeta` from a
   431→   431→  // later `const` would throw a TDZ ReferenceError.
   432→   432→  const activeFile = files.find((f) => f.id === activeId) ?? null
   433→   433→  const activeMeta = activeId ? meta[activeId] ?? null : null
   434→   434→  const activeRel = activeId ? relCrops[activeId] ?? null : null
   435→   435→
   436→   436→  /* ---------------- Container measurement -------------------------------- */
   437→   437→  const updateSize = React.useCallback(() => {
   438→   438→    const el = containerRef.current
   439→   439→    if (!el) return
   440→   440→    setContainerSize((prev) => {
   441→   441→      const w = el.clientWidth
   442→   442→      const h = el.clientHeight
   443→   443→      if (w === prev.w && h === prev.h) return prev
   444→   444→      return { w, h }
   445→   445→    })
   446→   446→  }, [])
   447→   447→
   448→   448→  // Measure synchronously before paint so the overlay is never stale after
   449→   449→  // switching images (this was the source of stretched/contracted rects).
   450→   450→  React.useLayoutEffect(() => {
   451→   451→    updateSize()
   452→   452→    const el = containerRef.current
   453→   453→    if (!el) return
   454→   454→    const ro = new ResizeObserver(updateSize)
   455→   455→    ro.observe(el)
   456→   456→    return () => ro.disconnect()
   457→   457→  }, [activeId, activeMeta, updateSize])
   458→   458→
   459→   459→  /* ---------------- Emit result whenever crops change -------------------- */
   460→   460→  const croppedCount = files.filter((f) => relCrops[f.id]).length
   461→   461→
   462→   462→  React.useEffect(() => {
   463→   463→    const crops: CropImagesResult['crops'] = {}
   464→   464→    for (const f of files) {
   465→   465→      const rel = relCrops[f.id]
   466→   466→      const m = meta[f.id]
   467→   467→      if (rel && m) {
   468→   468→        crops[f.file.name] = {
   469→   469→          x: Math.round(rel.x * m.width),
   470→   470→          y: Math.round(rel.y * m.height),
   471→   471→          width: Math.max(1, Math.round(rel.w * m.width)),
   472→   472→          height: Math.max(1, Math.round(rel.h * m.height)),
   473→   473→        }
   474→   474→      }
   475→   475→    }
   476→   476→    onChange(Object.keys(crops).length > 0 ? { crops } : null)
   477→   477→  }, [files, relCrops, meta, onChange])
   478→   478→
   479→   479→  const setRelFromPx = React.useCallback(
   480→   480→    (rect: PxRect | null, box: { w: number; h: number }) => {
   481→   481→      if (!activeId) return
   482→   482→      setRelCrops((prev) => {
   483→   483→        const next = { ...prev }
   484→   484→        if (!rect || rect.width <= 0 || rect.height <= 0) {
   485→   485→          delete next[activeId]
   486→   486→        } else {
   487→   487→          next[activeId] = {
   488→   488→            x: rect.x / box.w,
   489→   489→            y: rect.y / box.h,
   490→   490→            w: rect.width / box.w,
   491→   491→            h: rect.height / box.h,
   492→   492→          }
   493→   493→        }
   494→   494→        return next
   495→   495→      })
   496→   496→    },
   497→   497→    [activeId]
   498→   498→  )
   499→   499→
   500→   500→  const activePx: PxRect | null =
   501→   501→    activeRel && containerSize.w > 0 && containerSize.h > 0
   502→   502→      ? relToPx(activeRel, containerSize.w, containerSize.h)
   503→   503→      : null
   504→   504→
   505→   505→  /* ---------------- Pointer interaction ----------------------------------- */
   506→   506→  const posFromEvent = (e: React.PointerEvent): PxPoint => {
   507→   507→    const el = containerRef.current
   508→   508→    if (!el) return { x: 0, y: 0 }
   509→   509→    const r = el.getBoundingClientRect()
   510→   510→    // The container has a border; the overlay lives inside it. Subtract the
   511→   511→    // border so pointer coordinates share the exact space as clientWidth and
   512→   512→    // the absolutely positioned crop rect (otherwise every hit is offset).
   513→   513→    return {
   514→   514→      x: clamp(e.clientX - r.left - el.clientLeft, 0, el.clientWidth),
   515→   515→      y: clamp(e.clientY - r.top - el.clientTop, 0, el.clientHeight),
   516→   516→    }
   517→   517→  }
   518→   518→
   519→   519→  const onPointerDown = (e: React.PointerEvent) => {
   520→   520→    if (!activeMeta || e.button !== 0) return

   522→   522→    if (!el) return
   523→   523→    const box = { w: el.clientWidth, h: el.clientHeight }
   524→   524→    if (!box.w || !box.h) return
   525→   525→    // Capture the pointer: drags keep working even when the cursor leaves
   526→   526→    // the image, and the up-event can never be missed.
   527→   527→    try {
   528→   528→      el.setPointerCapture(e.pointerId)
   529→   529→    } catch {
   530→   530→      // ignore — capture is best-effort
   531→   531→    }

   533→   533→    const px = activeRel ? relToPx(activeRel, box.w, box.h) : null
   534→   534→    const hit = px ? hitTest(pos, px) : null
   535→   535→    // Lock the aspect ratio for the entire drag at pointer-down time.
   536→   536→    const dragRatio = ratioValue(ratio, activeMeta)
   537→   537→    if (hit && px) {
   538→   538→      // Safety net: a CORNER resize with a locked ratio must start from a
   539→   539→      // rect that is EXACTLY at that ratio (corner math derives height from
   540→   540→      // width, so an off-ratio start — e.g. after a single-axis mid-edge
   541→   541→      // drag — would jump on the first move). Mid-edge handles are strictly
   542→   542→      // single-axis and never consult the ratio, so they must NOT snap:
   543→   543→      // snapping would undo the user's one-axis adjustment at grab time.

   545→   545→        hit === 'resize-nw' ||
   546→   546→        hit === 'resize-ne' ||
   547→   547→        hit === 'resize-sw' ||
   548→   548→        hit === 'resize-se'

   551→   551→        const snapped = fitRectToRatio(px, dragRatio, box.w, box.h)
   552→   552→        if (
   553→   553→          Math.abs(snapped.width - px.width) > 0.5 ||
   554→   554→          Math.abs(snapped.height - px.height) > 0.5
   555→   555→        ) {
   556→   556→          setRelFromPx(snapped, box)
   557→   557→        }
   558→   558→        startRect = snapped
   559→   559→      }
   560→   560→      drag.current = {
   561→   561→        mode: hit,
   562→   562→        anchor: pos,
   563→   563→        start: startRect,
   564→   564→        box,
   565→   565→        ratio: dragRatio,
   566→   566→        lastRect: { ...startRect },
   567→   567→      }
   568→   568→    } else {
   569→   569→      const seed = { x: pos.x, y: pos.y, width: 0, height: 0 }
   570→   570→      drag.current = {
   571→   571→        mode: 'creating',
   572→   572→        anchor: pos,
   573→   573→        start: null,
   574→   574→        box,
   575→   575→        ratio: dragRatio,
   576→   576→        lastRect: seed,
   577→   577→      }
   578→   578→      setRelFromPx(seed, box)
   579→   579→    }
   580→   580→  }
   581→   581→
   582→   582→  const onPointerMove = (e: React.PointerEvent) => {

   584→   584→    if (!d) return

   586→   586→    const { w: cw, h: ch } = d.box

   589→   589→      d.lastRect = r

   591→   591→      return
   592→   592→    }

   594→   594→    const dx = pos.x - d.anchor.x
   595→   595→    const dy = pos.y - d.anchor.y
   596→   596→    if (d.mode === 'moving') {
   597→   597→      const x = clamp(d.start.x + dx, 0, Math.max(0, cw - d.start.width))
   598→   598→      const y = clamp(d.start.y + dy, 0, Math.max(0, ch - d.start.height))
   599→   599→      setRelFromPx({ ...d.start, x, y }, d.box)
   600→   600→      return
   601→   601→    }
   602→   602→    const next = resizeWithHandles(d.mode, d.start, dx, dy, cw, ch, d.ratio)
   603→   603→    d.lastRect = next
   604→   604→    setRelFromPx(next, d.box)
   605→   605→  }

   612→   612→      if (d.lastRect.width < MIN_SIZE || d.lastRect.height < MIN_SIZE) {
   613→   613→        setRelFromPx(null, d.box)
   614→   614→      }
   615→   615→    }
   616→   616→  }
   617→   617→
   618→   618→  /* ---------------- Toolbar actions --------------------------------------- */
   619→   619→  const applyRatioPreset = (r: Ratio) => {

   690→   690→            m.height

   717→   717→

   719→   719→

   722→   722→      {/* Instructions */}
