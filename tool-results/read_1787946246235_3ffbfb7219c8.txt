    59→    59→interface InputConfig {
    60→    60→  accept: string
    61→    61→  multiple: boolean
    62→    62→  hint: string
    63→    63→  mode: 'files' | 'text'
    64→    64→}
    65→    65→
    66→    66→function getInput(tool: Tool): InputConfig {
    67→    67→  switch (tool.id) {
    68→    68→    case 'crop-images':
    69→    69→      return { accept: 'image/jpeg,image/jpg,image/png,image/webp', multiple: true, hint: 'JPG, PNG, WEBP images', mode: 'files' }
    70→    70→    case 'images-to-pdf':
    71→    71→      return { accept: 'image/*', multiple: true, hint: 'JPG, PNG, WEBP, GIF', mode: 'files' }
    72→    72→    case 'word-to-pdf':
    73→    73→      return { accept: '.docx,application/vnd.openxmlformats-officedocument.wordprocessingml.document', multiple: true, hint: 'Word .docx files', mode: 'files' }
    74→    74→    case 'excel-to-pdf':
    75→    75→      return { accept: '.xlsx,application/vnd.openxmlformats-officedocument.spreadsheetml.sheet', multiple: true, hint: 'Excel .xlsx files', mode: 'files' }
    76→    76→    case 'html-to-pdf':
    77→    77→      return { accept: '', multiple: false, hint: 'Paste your HTML', mode: 'text' }
    78→    78→    default:
    79→    79→      return { accept: 'application/pdf', multiple: tool.batch, hint: 'PDF files', mode: 'files' }
    80→    80→  }
    81→    81→}

   377→   377→      }

   379→   379→      singleLabel = 'Word → PDF output'

   382→   382→      // (preserving colors, merges, borders, fonts), then send images to worker
